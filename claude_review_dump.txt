#pragma once

#include <cstdint>

namespace hlc {

using Hlc = std::uint64_t;
using PhysicalNowFn = std::uint64_t (*)();

constexpr std::uint8_t LOGICAL_BITS = 16;
constexpr std::uint16_t LOGICAL_MAX = 0xFFFFu;
constexpr std::uint64_t PHYSICAL_MASK = (1ULL << 48) - 1ULL;

constexpr std::uint64_t MAX_REMOTE_DRIFT_MS = 60000;

constexpr Hlc make(
    std::uint64_t physical_ms,
    std::uint16_t logical_counter
) {
    return ((physical_ms & PHYSICAL_MASK) << LOGICAL_BITS) |
           static_cast<Hlc>(logical_counter);
}

constexpr std::uint64_t physical(Hlc value) {
    return (value >> LOGICAL_BITS) & PHYSICAL_MASK;
}

constexpr std::uint16_t logical(Hlc value) {
    return static_cast<std::uint16_t>(value & LOGICAL_MAX);
}

int compare(Hlc lhs, Hlc rhs);

class Clock {
public:
    explicit Clock(PhysicalNowFn physical_now);

    bool begin(Hlc persisted_hlc);

    Hlc now() const;
    Hlc on_local_event();
    Hlc on_receive(Hlc remote_hlc);

    bool ready() const;

private:
    std::uint64_t physical_now_ms() const;
    Hlc wait_for_physical_advance(
        std::uint64_t previous_physical
    );

    PhysicalNowFn physical_now_;
    std::uint64_t physical_offset_ms_ = 0;
    Hlc value_ = 0;
    bool initialized_ = false;
};

} // namespace hlc

namespace consistent_hash {

std::uint32_t fnv1a(
    const std::uint8_t* data,
    std::size_t length
)
{
    if (data == nullptr && length != 0) {
        return 0;
    }

    std::uint32_t hash =
        FNV1A_OFFSET_BASIS;

    for (std::size_t i = 0; i < length; ++i) {
        hash ^= data[i];
        hash *= FNV1A_PRIME;
    }

    return hash;
}

} // namespace consistent_hash

#include "dedup_manager.h"

#include <cstring>

namespace request {

DedupManager::DedupManager(
    std::uint64_t retention_ms
)
    : retention_ms_(retention_ms)
{
}

bool DedupManager::begin()
{
    if (retention_ms_ == 0) {
        return false;
    }

    for (std::size_t i = 0;
         i < MAX_DEDUP_ENTRIES;
         ++i) {

        entries_[i] = Entry{};
    }

    initialized_ = true;

    return true;
}

int DedupManager::find(
    Identity identity
) const
{
    for (std::size_t i = 0;
         i < MAX_DEDUP_ENTRIES;
         ++i) {

        if (entries_[i].state ==
            DedupState::FREE) {

            continue;
        }

        if (entries_[i].identity == identity) {
            return static_cast<int>(i);
        }
    }

    return -1;
}

int DedupManager::find_free() const
{
    for (std::size_t i = 0;
         i < MAX_DEDUP_ENTRIES;
         ++i) {

        if (entries_[i].state ==
            DedupState::FREE) {

            return static_cast<int>(i);
        }
    }

    return -1;
}

bool DedupManager::expired(
    const Entry& entry,
    std::uint64_t now_ms
) const
{
    if (entry.state !=
        DedupState::COMPLETED) {

        return false;
    }

    if (now_ms <
        entry.completed_at_ms) {

        return false;
    }

    return (
        now_ms - entry.completed_at_ms
    ) >= retention_ms_;
}

DedupResult DedupManager::begin_request(
    Identity identity,
    std::uint64_t now_ms
)
{
    if (!initialized_) {
        return DedupResult::NOT_READY;
    }

    if (!identity.valid()) {
        return DedupResult::INVALID_ARGUMENT;
    }

    /*
     * First clean expired completed entries.
     */
    expire(now_ms);

    const int existing =
        find(identity);

    if (existing >= 0) {

        const Entry& entry =
            entries_[existing];

        if (entry.state ==
            DedupState::IN_PROGRESS) {

            return DedupResult::
                DUPLICATE_IN_PROGRESS;
        }

        if (entry.state ==
            DedupState::COMPLETED) {

            return DedupResult::
                DUPLICATE_COMPLETED;
        }
    }

    const int slot = find_free();

    if (slot < 0) {
        return DedupResult::TABLE_FULL;
    }

    Entry& entry = entries_[slot];

    entry = Entry{};

    entry.identity = identity;
    entry.state =
        DedupState::IN_PROGRESS;

    return DedupResult::NEW_REQUEST;
}

DedupResult DedupManager::complete_request(
    Identity identity,
    const std::uint8_t* response,
    std::size_t response_length,
    std::uint64_t now_ms
)
{
    if (!initialized_) {
        return DedupResult::NOT_READY;
    }

    if (!identity.valid()) {
        return DedupResult::INVALID_ARGUMENT;
    }

    if (response_length >
        MAX_RESPONSE_SIZE) {

        return DedupResult::BUFFER_TOO_SMALL;
    }

    if (response_length > 0 &&
        response == nullptr) {

        return DedupResult::INVALID_ARGUMENT;
    }

    expire(now_ms);

    const int slot =
        find(identity);

    if (slot < 0) {
        return DedupResult::UNKNOWN_REQUEST;
    }

    Entry& entry = entries_[slot];

    if (entry.state ==
        DedupState::COMPLETED) {

        return DedupResult::
            DUPLICATE_COMPLETED;
    }

    if (entry.state !=
        DedupState::IN_PROGRESS) {

        return DedupResult::
            UNKNOWN_REQUEST;
    }

    if (response_length > 0) {
        std::memcpy(
            entry.response,
            response,
            response_length
        );
    }

    entry.response_length =
        static_cast<std::uint8_t>(
            response_length
        );

    entry.completed_at_ms =
        now_ms;

    entry.state =
        DedupState::COMPLETED;

    return DedupResult::NEW_REQUEST;
}

DedupResult DedupManager::get_state(
    Identity identity,
    DedupState& state,
    std::uint64_t now_ms
)
{
    state = DedupState::FREE;

    if (!initialized_) {
        return DedupResult::NOT_READY;
    }

    if (!identity.valid()) {
        return DedupResult::INVALID_ARGUMENT;
    }

    expire(now_ms);

    const int slot =
        find(identity);

    if (slot < 0) {
        return DedupResult::UNKNOWN_REQUEST;
    }

    state =
        entries_[slot].state;

    if (state ==
        DedupState::IN_PROGRESS) {

        return DedupResult::
            DUPLICATE_IN_PROGRESS;
    }

    return DedupResult::
        DUPLICATE_COMPLETED;
}

DedupResult DedupManager::get_cached_response(
    Identity identity,
    std::uint8_t* output,
    std::size_t output_capacity,
    std::size_t& output_length,
    std::uint64_t now_ms
)
{
    output_length = 0;

    if (!initialized_) {
        return DedupResult::NOT_READY;
    }

    if (!identity.valid()) {
        return DedupResult::INVALID_ARGUMENT;
    }

    if (output == nullptr) {
        return DedupResult::INVALID_ARGUMENT;
    }

    expire(now_ms);

    const int slot =
        find(identity);

    if (slot < 0) {
        return DedupResult::UNKNOWN_REQUEST;
    }

    const Entry& entry =
        entries_[slot];

    if (entry.state !=
        DedupState::COMPLETED) {

        return DedupResult::
            DUPLICATE_IN_PROGRESS;
    }

    if (output_capacity <
        entry.response_length) {

        return DedupResult::BUFFER_TOO_SMALL;
    }

    if (entry.response_length > 0) {
        std::memcpy(
            output,
            entry.response,
            entry.response_length
        );
    }

    output_length =
        entry.response_length;

    return DedupResult::
        DUPLICATE_COMPLETED;
}

void DedupManager::expire(
    std::uint64_t now_ms
)
{
    if (!initialized_) {
        return;
    }

    for (std::size_t i = 0;
         i < MAX_DEDUP_ENTRIES;
         ++i) {

        Entry& entry =
            entries_[i];

        if (entry.state !=
            DedupState::COMPLETED) {

            continue;
        }

        if (expired(
                entry,
                now_ms
            )) {

            entry = Entry{};
        }
    }
}

DedupResult DedupManager::remove(
    Identity identity
)
{
    if (!initialized_) {
        return DedupResult::NOT_READY;
    }

    if (!identity.valid()) {
        return DedupResult::INVALID_ARGUMENT;
    }

    const int slot =
        find(identity);

    if (slot < 0) {
        return DedupResult::UNKNOWN_REQUEST;
    }

    entries_[slot] = Entry{};

    return DedupResult::DUPLICATE_COMPLETED;
}

std::size_t DedupManager::active_count() const
{
    std::size_t count = 0;

    if (!initialized_) {
        return 0;
    }

    for (std::size_t i = 0;
         i < MAX_DEDUP_ENTRIES;
         ++i) {

        if (entries_[i].state !=
            DedupState::FREE) {

            ++count;
        }
    }

    return count;
}

bool DedupManager::ready() const
{
    return initialized_;
}

std::uint64_t DedupManager::retention_ms() const
{
    return retention_ms_;
}

bool valid_range(
    std::uint16_t address,
    std::size_t length
) {
    if (length == 0) {
        return address <= SIZE_BYTES;
    }

    if (address > LAST_ADDRESS) {
        return false;
    }

    return length <=
        (static_cast<std::size_t>(SIZE_BYTES) - address);
}

Result wait_for_write_cycle() {
    const unsigned long start = millis();

    while ((millis() - start) < WRITE_CYCLE_TIMEOUT_MS) {
        Wire.beginTransmission(I2C_ADDRESS);

        const std::uint8_t error =
            Wire.endTransmission(true);

        if (error == 0) {
            return Result::OK;
        }

        delay(1);
    }

    return Result::WRITE_TIMEOUT;
}

    std::uint8_t scl_pin
) {
    initialized_ = false;

    if (!Wire.begin(sda_pin, scl_pin)) {
        return Result::I2C_ERROR;
    }

    Wire.setClock(I2C_FREQUENCY_HZ);

    Wire.beginTransmission(I2C_ADDRESS);

    const std::uint8_t error =
        Wire.endTransmission(true);

    if (error != 0) {
        return Result::I2C_ERROR;
    }

    initialized_ = true;

    return Result::OK;
}

Result AT24C256::read(
    std::uint16_t address,
    std::uint8_t* buffer,
    std::size_t length
) {
    if (!initialized_) {
        return Result::NOT_INITIALIZED;
    }

    if (length > 0 && buffer == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (!valid_range(address, length)) {
        return Result::OUT_OF_RANGE;
    }

    if (length == 0) {
        return Result::OK;
    }

    std::size_t remaining = length;
    std::uint16_t current_address = address;
    std::uint8_t* current_buffer = buffer;

    while (remaining > 0) {
        const std::size_t chunk =
            (remaining < READ_CHUNK_SIZE)
                ? remaining
                : READ_CHUNK_SIZE;

        Wire.beginTransmission(I2C_ADDRESS);

        Wire.write(
            static_cast<std::uint8_t>(
                current_address >> 8
            )
        );

        Wire.write(
            static_cast<std::uint8_t>(
                current_address & 0xFF
            )
        );

        const std::uint8_t address_result =
            Wire.endTransmission(false);

        if (address_result != 0) {
            return Result::I2C_ERROR;
        }

        const std::size_t received =
    Wire.requestFrom(
        static_cast<std::uint8_t>(I2C_ADDRESS),
        static_cast<std::size_t>(chunk),
        true
    );

        if (received != chunk) {
            return Result::I2C_ERROR;
        }

        for (std::size_t i = 0; i < chunk; ++i) {
            if (!Wire.available()) {
                return Result::I2C_ERROR;
            }

            current_buffer[i] = Wire.read();
        }

        current_address =
            static_cast<std::uint16_t>(
                current_address + chunk
            );

        current_buffer += chunk;
        remaining -= chunk;
    }

    return Result::OK;
}

Result AT24C256::write(
    std::uint16_t address,
    const std::uint8_t* data,
    std::size_t length
) {
    if (!initialized_) {
        return Result::NOT_INITIALIZED;
    }

    if (length > 0 && data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (!valid_range(address, length)) {
        return Result::OUT_OF_RANGE;
    }

    if (length == 0) {
        return Result::OK;
    }

    std::size_t remaining = length;
    std::uint16_t current_address = address;
    const std::uint8_t* current_data = data;

    while (remaining > 0) {
        const std::size_t page_offset =
            current_address % PAGE_SIZE;

        const std::size_t bytes_to_page_end =
            PAGE_SIZE - page_offset;

        const std::size_t chunk =
            (remaining < bytes_to_page_end)
                ? remaining
                : bytes_to_page_end;

        Wire.beginTransmission(I2C_ADDRESS);

        Wire.write(
            static_cast<std::uint8_t>(
                current_address >> 8
            )
        );

        Wire.write(
            static_cast<std::uint8_t>(
                current_address & 0xFF
            )
        );

        const std::size_t written =
            Wire.write(current_data, chunk);

        if (written != chunk) {
            return Result::I2C_ERROR;
        }

        const std::uint8_t write_result =
            Wire.endTransmission(true);

        if (write_result != 0) {
            return Result::I2C_ERROR;
        }

        const Result wait_result =
            wait_for_write_cycle();

        if (wait_result != Result::OK) {
            return wait_result;
        }

        current_address =
            static_cast<std::uint16_t>(
                current_address + chunk
            );

        current_data += chunk;
        remaining -= chunk;
    }

    return Result::OK;
}

bool AT24C256::ready() const {
    return initialized_;
}

std::uint32_t mix32(std::uint32_t value)
{
    value ^= value >> 16;
    value *= 0x7feb352dU;
    value ^= value >> 15;
    value *= 0x846ca68bU;
    value ^= value >> 16;

    return value;
}

std::uint32_t virtual_point_hash(
    node::Id id,
    std::uint8_t virtual_index
)
{
    const std::uint8_t seed[] = {
        id,
        virtual_index,
        0xC3,
        0x28
    };

    return mix32(
        consistent_hash::fnv1a(
            seed,
            sizeof(seed)
        )
    );
}

bool Ring::less(
    const VirtualPoint& lhs,
    const VirtualPoint& rhs
)
{
    if (lhs.position != rhs.position) {
        return lhs.position < rhs.position;
    }

    if (lhs.owner != rhs.owner) {
        return lhs.owner < rhs.owner;
    }

    return lhs.virtual_index < rhs.virtual_index;
}

bool Ring::begin()
{
    std::size_t position = 0;

    for (
        node::Id id = node::MIN;
        id <= node::MAX;
        ++id
    ) {
        for (
            std::uint8_t virtual_index = 0;
            virtual_index < VIRTUAL_NODES_PER_NODE;
            ++virtual_index
        ) {
            points_[position++] = VirtualPoint{
                virtual_point_hash(
                    id,
                    virtual_index
                ),
                id,
                virtual_index
            };
        }
    }

    for (
        std::size_t i = 1;
        i < RING_POINT_COUNT;
        ++i
    ) {
        const VirtualPoint current =
            points_[i];

        std::size_t j = i;

        while (
            j > 0 &&
            less(
                current,
                points_[j - 1]
            )
        ) {
            points_[j] =
                points_[j - 1];

            --j;
        }

        points_[j] = current;
    }

    initialized_ = true;

    return true;
}

bool Ring::ready() const
{
    return initialized_;
}

std::size_t Ring::point_count() const
{
    return RING_POINT_COUNT;
}

const VirtualPoint& Ring::point(
    std::size_t index
) const
{
    return points_[index];
}

bool Ring::find_successor(
    std::uint32_t hash,
    VirtualPoint& result
) const
{
    std::size_t index = 0;

    if (!find_successor_index(hash, index)) {
        return false;
    }

    result = points_[index];

    return true;
}

bool Ring::find_successor_index(
    std::uint32_t hash,
    std::size_t& result_index
) const
{
    if (!initialized_) {
        return false;
    }

    std::size_t left = 0;
    std::size_t right = RING_POINT_COUNT;

    while (left < right) {
        const std::size_t middle =
            left + (right - left) / 2;

        if (points_[middle].position < hash) {
            left = middle + 1;
        } else {
            right = middle;
        }
    }

    if (left == RING_POINT_COUNT) {
        result_index = 0;
    } else {
        result_index = left;
    }

    return true;
}

    std::uint8_t buffer[
        HEARTBEAT_BUFFER_SIZE
    ]{};

    for (;;) {

        const hlc::Hlc timestamp =
            clock_.now();

        if (timestamp != 0) {

            std::size_t bytes_written = 0;

            const protocol::Result build_result =
                heartbeat::build_packet(
                    local_node_id_,
                    timestamp,
                    buffer,
                    sizeof(buffer),
                    bytes_written
                );

            if (build_result == protocol::Result::OK) {

                const transport::Result send_result =
                    transport_.send_broadcast(
                        buffer,
                        bytes_written
                    );

                if (send_result == transport::Result::OK) {
                    Serial.println(
                        "HEARTBEAT TX: OK"
                    );
                } else {
                    Serial.print(
                        "HEARTBEAT TX: FAIL ("

                    Serial.println(")");

                );
                Serial.print(
                    static_cast<std::uint8_t>(
                        build_result
                    )
                );
                Serial.println(")");
            }

namespace hlc {

int compare(Hlc lhs, Hlc rhs)
{
    if (lhs < rhs) {
        return -1;
    }

    if (lhs > rhs) {
        return 1;
    }

    return 0;
}

Clock::Clock(PhysicalNowFn physical_now)
    : physical_now_(physical_now)
{
}

bool Clock::begin(Hlc persisted_hlc)
{
    if (physical_now_ == nullptr) {
        return false;
    }

    const std::uint64_t raw_now =
        physical_now_() & PHYSICAL_MASK;

    const std::uint64_t persisted_physical =
        physical(persisted_hlc);

    /*
     * The ESP32 monotonic timer restarts after reboot.
     *
     * If the persisted physical component is ahead of the
     * new boot's raw timer, create a runtime offset so that
     * the reconstructed physical clock starts at the
     * persisted value and continues from there.
     */
    if (raw_now <= persisted_physical) {
        physical_offset_ms_ =
            persisted_physical - raw_now;

        value_ = persisted_hlc;
    } else {
        /*
         * The new boot's monotonic time is already ahead of
         * the persisted physical component.
         */
        physical_offset_ms_ = 0;
        value_ = make(raw_now, 0);
    }

    initialized_ = true;
    return true;
}

std::uint64_t Clock::physical_now_ms() const
{
    const std::uint64_t raw_now =
        physical_now_() & PHYSICAL_MASK;

    if (raw_now > PHYSICAL_MASK - physical_offset_ms_) {
        return PHYSICAL_MASK;
    }

    return raw_now + physical_offset_ms_;
}

Hlc Clock::now() const
{
    if (!initialized_) {
        return 0;
    }

    const std::uint64_t current_physical =
        physical_now_ms();

    const std::uint64_t stored_physical =
        physical(value_);

    /*
     * now() is a read-only snapshot.
     * It does not itself count as a local event.
     */
    if (current_physical > stored_physical) {
        return make(current_physical, 0);
    }

    return value_;
}

Hlc Clock::wait_for_physical_advance(
    std::uint64_t previous_physical
)
{
    // Logical counter overflowed: don't block, just move
    // physical forward by 1 ms and reset the counter.
    value_ = make(previous_physical + 1, 0);
    return value_;
}

Hlc Clock::on_local_event()
{
    if (!initialized_) {
        return 0;
    }

    const std::uint64_t previous_physical =
        physical(value_);

    const std::uint16_t previous_logical =
        logical(value_);

    const std::uint64_t current_physical =
        physical_now_ms();


    const std::uint64_t next_physical =
        current_physical > previous_physical
            ? current_physical
            : previous_physical;

    /*
     * Physical time advanced.
     */
    if (next_physical > previous_physical) {
        value_ = make(next_physical, 0);
        return value_;
    }

    /*
     * Same physical millisecond:
     * increment logical component.
     */
    if (previous_logical == LOGICAL_MAX) {
        return wait_for_physical_advance(
            previous_physical
        );
    }

    value_ = make(
        previous_physical,
        static_cast<std::uint16_t>(
            previous_logical + 1u
        )
    );

    return value_;
}

Hlc Clock::on_receive(Hlc remote_hlc)
{
    if (!initialized_) {
        return 0;
    }

    const std::uint64_t local_physical =
        physical(value_);

    const std::uint16_t local_logical =
        logical(value_);

    const std::uint64_t remote_physical =
        physical(remote_hlc);

    const std::uint16_t remote_logical =
        logical(remote_hlc);

    const std::uint64_t current_physical =
        physical_now_ms();

    

if (remote_physical > current_physical &&
    remote_physical - current_physical >
        MAX_REMOTE_DRIFT_MS) {

    return 0;
}
    /*
     * p = max(local physical, remote physical, physical_now)
     */
    std::uint64_t next_physical = local_physical;

    if (remote_physical > next_physical) {
        next_physical = remote_physical;
    }

    if (current_physical > next_physical) {
        next_physical = current_physical;
    }

    std::uint16_t next_logical = 0;

    /*
     * p == local == remote
     */
    if (next_physical == local_physical &&
        next_physical == remote_physical) {

        const std::uint16_t highest_logical =
            local_logical > remote_logical
                ? local_logical
                : remote_logical;

        if (highest_logical == LOGICAL_MAX) {
            return wait_for_physical_advance(
                next_physical
            );
        }

        next_logical = static_cast<std::uint16_t>(
            highest_logical + 1u
        );

    /*
     * p == local
     */
    } else if (next_physical == local_physical) {

        if (local_logical == LOGICAL_MAX) {
            return wait_for_physical_advance(
                next_physical
            );
        }

        next_logical = static_cast<std::uint16_t>(
            local_logical + 1u
        );

    /*
     * p == remote
     */
    } else if (next_physical == remote_physical) {

        if (remote_logical == LOGICAL_MAX) {
            return wait_for_physical_advance(
                next_physical
            );
        }

        next_logical = static_cast<std::uint16_t>(
            remote_logical + 1u
        );

    /*
     * p == physical_now and is greater than both.
     */
    } else {
        next_logical = 0;
    }

    value_ = make(
        next_physical,
        next_logical
    );

    return value_;
}

bool Clock::ready() const
{
    return initialized_;
}

Result Engine::execute(
    const protocol::DecodedMessage& message,
    const membership::Manager& membership,
    std::uint64_t now_ms,
    OperationResponse& response
) const
{
    response = OperationResponse{};

    if (!ready() || !membership.ready()) {
        response.result = Result::NOT_READY;
        return response.result;
    }

    protocol::KeyView key{};

    switch (message.header.msg_type) {
        case protocol::MessageType::SET_REQ:
            key = message.payload.set_request.key;
            break;

        case protocol::MessageType::GET_REQ:
            key = message.payload.get_request.key;
            break;

        case protocol::MessageType::DEL_REQ:
            key = message.payload.delete_request.key;
            break;

        default:
            response.result = Result::INVALID_REQUEST;
            return response.result;
    }

    if (key.len > storage::KEY_MAX) {
        response.result = Result::KEY_TOO_LARGE;
        return response.result;
    }

    placement::Assignment assignment{};

    if (
        !placement_.locate(
            key.data,
            key.len,
            membership,
            now_ms,
            assignment
        )
    ) {
        response.result = Result::NOT_RESPONSIBLE;
        copy_key(key, response);
        return response.result;
    }

    if (assignment.primary != local_node_id_) {
        response.result = Result::NOT_RESPONSIBLE;
        copy_key(key, response);
        return response.result;
    }

    switch (message.header.msg_type) {
        case protocol::MessageType::SET_REQ:
            return execute_set(message, response);

        case protocol::MessageType::GET_REQ:
            return execute_get(message, response);

        case protocol::MessageType::DEL_REQ:
            return execute_delete(message, response);

        default:
            response.result = Result::INVALID_REQUEST;
            return response.result;
    }
}

Result Engine::execute_set(
    const protocol::DecodedMessage& message,
    OperationResponse& response
) const
{
    const protocol::KeyView key =
        message.payload.set_request.key;

    const protocol::ValueView value =
        message.payload.set_request.value;

    if (key.len > storage::KEY_MAX) {
        response.result = Result::KEY_TOO_LARGE;
        return response.result;
    }

    if (value.len > storage::VALUE_MAX) {
        response.result = Result::VALUE_TOO_LARGE;
        return response.result;
    }

    copy_key(key, response);

    const storage::Version version{
        message.header.hlc,
        message.header.origin_node_id
    };

    const storage::Result result =
        storage_.set(
            storage::KeyView{
                key.data,
                key.len
            },
            storage::ValueView{
                value.data,
                value.len
            },
            version
        );

    response.result = map_storage_result(result);

    return response.result;
}

Result Engine::execute_get(
    const protocol::DecodedMessage& message,
    OperationResponse& response
) const
{
    const protocol::KeyView key =
        message.payload.get_request.key;

    if (key.len > storage::KEY_MAX) {
        response.result = Result::KEY_TOO_LARGE;
        return response.result;
    }

    copy_key(key, response);

    std::size_t value_length = 0;

    const storage::Result result =
        storage_.get(
            storage::KeyView{
                key.data,
                key.len
            },
            response.value,
            sizeof(response.value),
            value_length
        );

    response.result = map_storage_result(result);

    if (response.result == Result::OK) {
        response.value_length =
            static_cast<std::uint16_t>(
                value_length
            );
    } else {
        response.value_length = 0;
    }

    return response.result;
}

Result Engine::execute_delete(
    const protocol::DecodedMessage& message,
    OperationResponse& response
) const
{
    const protocol::KeyView key =
        message.payload.delete_request.key;

    if (key.len > storage::KEY_MAX) {
        response.result = Result::KEY_TOO_LARGE;
        return response.result;
    }

    copy_key(key, response);

    const storage::Version version{
        message.header.hlc,
        message.header.origin_node_id
    };

    const storage::Result result =
        storage_.remove(
            storage::KeyView{
                key.data,
                key.len
            },

        );

    response.result = map_storage_result(result);

}

        case storage::Result::OK:
            return Result::OK;

        case storage::Result::NOT_FOUND:
            return Result::NOT_FOUND;

        case storage::Result::STALE_VERSION:
            return Result::STALE_VERSION;

        case storage::Result::KEY_TOO_LARGE:
            return Result::KEY_TOO_LARGE;

        case storage::Result::VALUE_TOO_LARGE:
            return Result::VALUE_TOO_LARGE;

        case storage::Result::NOT_INITIALIZED:
            return Result::NOT_READY;

    );
}

            return protocol::StatusCode::VALUE_TOO_LARGE;

        case Result::STORAGE_ERROR:
            return protocol::StatusCode::STORAGE_ERROR;

        case Result::NOT_RESPONSIBLE:
            return protocol::StatusCode::NOT_RESPONSIBLE;

        case Result::STALE_VERSION:
            return protocol::StatusCode::STALE_VERSION;

        case Result::NOT_READY:
            return protocol::StatusCode::STORAGE_ERROR;

        default:
            return protocol::StatusCode::INVALID_REQUEST;
    }
}

bool ResponseBuilder::build_payload(
    protocol::MessageType response_type,
    const OperationResponse& response,
    std::uint8_t* output,
    std::size_t output_capacity,
    std::size_t& payload_length
)
{
    payload_length = 0;

    const protocol::StatusCode status =
        to_protocol_status(response.result);

    const protocol::KeyView key{
        response.key,
        response.key_length
    };

    switch (response_type) {
        case protocol::MessageType::SET_ACK: {
            const protocol::SetAckPayload payload{
                status,
                key
            };

            return protocol::serialize_set_ack(
                payload,
                output,
                output_capacity,
                payload_length
            ) == protocol::Result::OK;
        }

        case protocol::MessageType::GET_RESP: {
            const protocol::ValueView value{
                response.value,
                static_cast<std::uint8_t>(
                    response.value_length
                )
            };

            const protocol::GetResponsePayload payload{
                status,
                key,
                value
            };

            return protocol::serialize_get_response(
                payload,
                output,
                output_capacity,
                payload_length
            ) == protocol::Result::OK;
        }

        case protocol::MessageType::DEL_ACK: {
            const protocol::DeleteAckPayload payload{
                status,
                key
            };

            return protocol::serialize_delete_ack(
                payload,
                output,
                output_capacity,
                payload_length
            ) == protocol::Result::OK;
        }

        default:
            return false;
    }
}

bool ResponseBuilder::build(
    const protocol::PacketHeader& request_header,
    node::Id local_node_id,
    protocol::Hlc response_hlc,
    const OperationResponse& response,
    std::uint8_t* output,
    std::size_t output_capacity,
    std::size_t& output_length
)
{
    output_length = 0;

    if (
        output == nullptr ||
        output_capacity < protocol::HEADER_SIZE
    ) {
        return false;
    }

    if (
        local_node_id < node::MIN ||
        local_node_id > node::MAX
    ) {
        return false;
    }

    const protocol::MessageType response_type =
        response_type_for(
            request_header.msg_type
        );

    if (
        response_type != protocol::MessageType::SET_ACK &&
        response_type != protocol::MessageType::GET_RESP &&
        response_type != protocol::MessageType::DEL_ACK
    ) {
        return false;
    }

    std::uint8_t payload[
        protocol::MAX_PAYLOAD_SIZE
    ]{};

    std::size_t payload_length = 0;

    if (
        !build_payload(
            response_type,
            response,
            payload,
            sizeof(payload),
            payload_length
        )
    ) {
        return false;
    }

    if (
        payload_length >
        protocol::MAX_PAYLOAD_SIZE
    ) {
        return false;
    }

    const protocol::PacketHeader response_header{
        protocol::PROTOCOL_VERSION,
        response_type,
        0,
        protocol::HEADER_SIZE,

        request_header.request_id,

        request_header.origin_node_id,
        local_node_id,
        request_header.origin_node_id,

        0,

        response_hlc,

        static_cast<std::uint16_t>(
            payload_length
        )
    };

    return protocol::serialize_packet(
        response_header,
        protocol::ByteView{
            payload,
            static_cast<std::uint8_t>(
                payload_length
            )
        },
        output,
        output_capacity,
        output_length
    ) == protocol::Result::OK;
}

bool build_set_request(
    request::Identity identity,
    const std::uint8_t* key,
    std::size_t key_length,
    const std::uint8_t* value,
    std::size_t value_length,
    std::uint8_t* output,
    std::size_t output_capacity,
    std::size_t& output_length
)

    if (
        key == nullptr ||
        value == nullptr ||
        output == nullptr

        key_length > storage::KEY_MAX) {

    }

    if (value_length > storage::VALUE_MAX) {
        return false;
    }

    protocol::SetRequestPayload payload{
        protocol::KeyView{
            key,

        protocol::ValueView{
            value,
            static_cast<std::uint8_t>(value_length)
        }
    };

    std::uint8_t payload_buffer[
        protocol::MAX_PAYLOAD_SIZE

        protocol::PROTOCOL_VERSION,
        protocol::MessageType::SET_REQ,
        0,
        protocol::HEADER_SIZE,

        identity.request_id,

        TEST_PEER_NODE_ID,

    };

    return
        protocol::serialize_packet(
            header,
            protocol::ByteView{
                payload_buffer,
                payload_length

            output_capacity,
            output_length

bool reliable_send(
    void* context,
    node::Id destination,
    const std::uint8_t* packet,
    std::size_t packet_length,
    bool retry
)
{
    auto* sender =
        static_cast<RequestSenderContext*>(
            context
        );

    if (
        sender == nullptr ||
        sender->transport == nullptr ||
        packet == nullptr
    ) {
        return false;
    }

    if (
        destination < node::MIN ||
        destination > node::MAX
    ) {
        return false;
    }

    if (!retry) {

        return
            sender->transport->send_unicast(
                destination,
                packet,
                packet_length
            ) == transport::Result::OK;
    }

    std::uint8_t retry_packet[
        protocol::MAX_PACKET_SIZE
    ]{};

    std::size_t retry_length = 0;

    if (
        request::prepare_retry(
            packet,
            packet_length,
            retry_packet,
            sizeof(retry_packet),
            retry_length
        ) != request::PacketResult::OK
    ) {
        return false;
    }

    Serial.print("RETRY SEND -> NODE ");
    Serial.println(
        static_cast<unsigned>(destination)
    );

    return
        sender->transport->send_unicast(
            destination,
            retry_packet,
            retry_length
        ) == transport::Result::OK;
}

bool find_key_owned_by_node(
    node::Id desired_primary
)
{
    /*
     * The test uses only A and C as live nodes.
     * This gives both boards the same routing view.
     */

    const std::uint32_t boot_tag = esp_random();

    for (unsigned index = 0; index < 1000; ++index) {

        const int written = std::snprintf(
            reinterpret_cast<char*>(test_key),
            sizeof(test_key),
            "phase6_wire_%lu_%03u",
            static_cast<unsigned long>(boot_tag),
            index
        );

        if (written <= 0) {
            continue;
        }

        const std::size_t length =
            static_cast<std::size_t>(written);

        if (length > storage::KEY_MAX) {
            continue;
        }

        placement::Assignment assignment{};

        if (
            !placement_resolver.locate(
                test_key,
                length,
                membership_manager,
                static_cast<std::uint64_t>(millis()),
                assignment
            )
        ) {
            continue;
        }

        if (assignment.primary != desired_primary) {
            continue;
        }

        test_key_length = length;

        Serial.print("TEST KEY: ");

        for (std::size_t i = 0; i < test_key_length; ++i) {
            Serial.write(test_key[i]);
        }

        Serial.println();

        Serial.print("PRIMARY: ");
        Serial.println(
            static_cast<unsigned>(assignment.primary)
        );

        Serial.print("REPLICA: ");
        Serial.println(
            static_cast<unsigned>(assignment.replica)
        );

        return true;
    }

    return false;
}

void process_server_request(
    const transport::ReceivedFrame& frame
)
{
    protocol::PacketView packet{};

    if (
        protocol::deserialize_packet(
            frame.data,
            frame.length,
            packet
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "RX PACKET DECODE: FAIL"
        );
        return;
    }

    if (
        packet.header.destination_node_id !=
        LOCAL_NODE_ID
    ) {
        return;
    }

    if (
        packet.header.msg_type !=
            protocol::MessageType::SET_REQ &&
        packet.header.msg_type !=
            protocol::MessageType::GET_REQ &&
        packet.header.msg_type !=
            protocol::MessageType::DEL_REQ
    ) {
        return;
    }

    protocol::DecodedMessage decoded{};

    if (
        protocol::decode_message(
            packet,
            decoded
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "APPLICATION DECODE: FAIL"
        );
        return;
    }

    request::Identity identity{};

    if (
        request::extract_identity(
            packet,
            identity
        ) != request::PacketResult::OK
    ) {
        Serial.println(
            "REQUEST IDENTITY: FAIL"
        );
        return;
    }

    /*
     * Merge remote HLC before application processing.
     */
    const hlc::Hlc merged_hlc =
        hlc_clock.on_receive(
            packet.header.hlc
        );

    if (merged_hlc == 0) {
        Serial.println(
            "RX HLC MERGE: REJECTED"
        );
        return;
    }

    const std::uint64_t now =
        static_cast<std::uint64_t>(millis());

    const request::DedupResult dedup_result =
        dedup_manager.begin_request(
            identity,
            now
        );

    if (
        dedup_result ==
        request::DedupResult::NEW_REQUEST
    ) {

        Serial.println();
        Serial.println(
            "SERVER: NEW REQUEST"
        );

        kv::OperationResponse operation_response{};

        const kv::Result operation_result =
            kv_engine.execute(
                decoded,
                membership_manager,
                now,
                operation_response
            );

        Serial.print(
            "KV ENGINE RESULT: "
        );

        Serial.println(
            static_cast<unsigned>(
                operation_result
            )
        );

        if (
            packet.header.msg_type ==
            protocol::MessageType::SET_REQ
        ) {
            ++receiver_set_execution_count;

            Serial.print(
                "SET EXECUTION COUNT: "
            );

            Serial.println(
                receiver_set_execution_count
            );
        }

        std::uint8_t response[
            protocol::MAX_PACKET_SIZE
        ]{};

        std::size_t response_length = 0;

        const hlc::Hlc response_hlc =
            hlc_clock.on_local_event();

        if (
            !kv::ResponseBuilder::build(
                packet.header,
                LOCAL_NODE_ID,
                response_hlc,
                operation_response,
                response,
                sizeof(response),
                response_length
            )
        ) {
            Serial.println(
                "RESPONSE BUILD: FAIL"
            );
            return;
        }

        const request::DedupResult cache_result =
            dedup_manager.complete_request(
                identity,
                response,
                response_length,
                now
            );

        if (
            cache_result !=
            request::DedupResult::NEW_REQUEST
        ) {
            Serial.print(
                "DEDUP CACHE: FAIL ("
            );

            Serial.print(
                static_cast<unsigned>(
                    cache_result
                )
            );

            Serial.println(")");

            return;
        }

        Serial.println(
            "DEDUP CACHE: COMPLETED"
        );

        /*
         * Intentionally lose the first SET_ACK.
         */
        if (
            packet.header.msg_type ==
                protocol::MessageType::SET_REQ &&
            drop_first_set_ack
        ) {
            drop_first_set_ack = false;

            Serial.println(
                "SET_ACK: DELIBERATELY DROPPED"
            );

            return;
        }

        const transport::Result send_result =
            espnow_transport.send_unicast(
                packet.header.origin_node_id,
                response,
                response_length
            );

        if (
            send_result ==
            transport::Result::OK
        ) {
            Serial.println(
                "RESPONSE: SENT"
            );
        } else {
            Serial.print(
                "RESPONSE SEND: FAIL ("
            );

            Serial.print(
                static_cast<unsigned>(
                    send_result
                )
            );

            Serial.println(")");
        }

        return;
    }

    if (
        dedup_result ==
        request::DedupResult::DUPLICATE_IN_PROGRESS
    ) {

        Serial.println(
            "DEDUP: DUPLICATE IN_PROGRESS - SUPPRESSED"
        );

        return;
    }

    if (
        dedup_result ==
        request::DedupResult::DUPLICATE_COMPLETED
    ) {

        Serial.println(
            "DEDUP: DUPLICATE_COMPLETED - REPLAY"
        );

        std::uint8_t cached_response[
            protocol::MAX_PACKET_SIZE
        ]{};

        std::size_t cached_length = 0;

        const request::DedupResult cache_result =
            dedup_manager.get_cached_response(
                identity,
                cached_response,
                sizeof(cached_response),
                cached_length,
                now
            );

        if (
            cache_result !=
            request::DedupResult::DUPLICATE_COMPLETED
        ) {
            Serial.println(
                "DEDUP CACHE READ: FAIL"
            );
            return;
        }

        const transport::Result send_result =
            espnow_transport.send_unicast(
                packet.header.origin_node_id,
                cached_response,
                cached_length
            );

        if (
            send_result ==
            transport::Result::OK
        ) {
            Serial.println(
                "CACHED RESPONSE: REPLAYED"
            );
        } else {
            Serial.print(
                "CACHED RESPONSE SEND: FAIL ("
            );

            Serial.print(
                static_cast<unsigned>(
                    send_result
                )
            );

            Serial.println(")");
        }

        return;
    }

    Serial.print(
        "DEDUP: UNEXPECTED RESULT ("
    );

    Serial.print(
        static_cast<unsigned>(
            dedup_result
        )
    );

    Serial.println(")");
}

void process_requester_response(
    const transport::ReceivedFrame& frame
)
{
    protocol::PacketView packet{};

    if (
        protocol::deserialize_packet(
            frame.data,
            frame.length,
            packet
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "RESPONSE PACKET DECODE: FAIL"
        );
        return;
    }

    if (
        packet.header.destination_node_id !=
        LOCAL_NODE_ID
    ) {
        return;
    }

    if (
        packet.header.msg_type !=
        protocol::MessageType::SET_ACK
    ) {
        return;
    }

    protocol::DecodedMessage decoded{};

    if (
        protocol::decode_message(
            packet,
            decoded
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "SET_ACK DECODE: FAIL"
        );
        return;
    }

    const hlc::Hlc merged_hlc =
        hlc_clock.on_receive(
            packet.header.hlc
        );

    if (merged_hlc == 0) {
        Serial.println(
            "ACK HLC MERGE: REJECTED"
        );
        return;
    }

    request::Identity identity{
        packet.header.origin_node_id,
        packet.header.request_id
    };

    if (!identity.valid()) {
        Serial.println(
            "ACK IDENTITY: FAIL"
        );
        return;
    }

    const request::Result completion_result =
        request_manager.complete(
            identity,
            frame.data,
            frame.length
        );

    if (
        completion_result ==
        request::Result::OK
    ) {
        request_finished = true;

        Serial.println();
        Serial.println(
            "REQUEST COMPLETION: PASS"
        );

        return;
    }

    if (
        completion_result ==
            request::Result::UNKNOWN_REQUEST ||
        completion_result ==
            request::Result::ALREADY_TERMINAL
    ) {
        Serial.println(
            "RESPONSE: LATE/UNKNOWN - IGNORED"
        );
        return;
    }

    Serial.print(
        "REQUEST COMPLETION: FAIL ("
    );

    Serial.print(
        static_cast<unsigned>(
            completion_result
        )
    );

    Serial.println(")");
}

void print_final_result()
{
    if (
        final_result_printed ||
        !request_finished
    ) {
        return;
    }

    final_result_printed = true;

    request::State state{};

    if (
        request_manager.get_state(
            active_request,
            state
        ) != request::Result::OK
    ) {
        Serial.println(
            "FINAL STATE READ: FAIL"
        );
        return;
    }

    request::RequestInfo info{};

    if (
        request_manager.get_info(
            active_request,
            info
        ) != request::Result::OK
    ) {
        Serial.println(
            "FINAL INFO READ: FAIL"
        );
        return;
    }

    std::uint8_t response[
        protocol::MAX_PACKET_SIZE
    ]{};

    std::size_t response_length = 0;

    if (
        request_manager.get_completion(
            active_request,
            response,
            sizeof(response),
            response_length
        ) != request::Result::OK
    ) {
        Serial.println(
            "FINAL RESPONSE READ: FAIL"
        );
        return;
    }

    protocol::PacketView packet{};

    if (
        protocol::deserialize_packet(
            response,
            response_length,
            packet
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "FINAL RESPONSE PACKET: FAIL"
        );
        return;
    }

    protocol::DecodedMessage decoded{};

    if (
        protocol::decode_message(
            packet,
            decoded
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "FINAL RESPONSE DECODE: FAIL"
        );
        return;
    }

    const bool status_ok =
        decoded.payload.set_ack.status ==
        protocol::StatusCode::OK;

    const bool retry_happened =
        info.retry_count > 0;

    const bool receiver_executed_once =
        receiver_set_execution_count == 1;

    Serial.println();
    Serial.println(
        "========================================"
    );
    Serial.println(
        "PHASE 6 DISTRIBUTED SET RESULT"
    );
    Serial.println(
        "========================================"
    );

    Serial.print(
        "FINAL REQUEST STATE: "
    );

    Serial.println(
        state == request::State::COMPLETED
            ? "COMPLETED"
            : "NOT COMPLETED"
    );

    Serial.print(
        "SET_ACK STATUS: "
    );

    Serial.println(
        status_ok
            ? "OK"
            : "NOT OK"
    );

    Serial.print(
        "RETRY COUNT: "
    );

    Serial.println(
        static_cast<unsigned>(
            info.retry_count
        )
    );

    Serial.print(
        "RETRY PATH: "
    );

    Serial.println(
        retry_happened
            ? "EXERCISED"
            : "NOT EXERCISED"
    );

    Serial.println(
        "RECEIVER EXECUTION COUNT MUST BE 1"
    );

    if (
        state == request::State::COMPLETED &&
        status_ok &&
        retry_happened &&
        receiver_executed_once
    ) {
        Serial.println();
        Serial.println(
            "PHASE 6 DISTRIBUTED SET: PASS"
        );
    } else {
        Serial.println();
        Serial.println(
            "PHASE 6 DISTRIBUTED SET: FAIL"
        );
    }

    Serial.println(
        "========================================"
    );
}

void start_client_request()
{
    if (
    LOCAL_NODE_ID != node::C ||
    request_started
) {
    return;
}

    /*
     * C must choose a key whose current primary is A.
     */
    if (
        !find_key_owned_by_node(
            node::A
        )
    ) {
        Serial.println(
            "NO KEY FOUND WITH PRIMARY A"
        );
        request_started = true;
        return;
    }

    static const std::uint8_t value[] = {
        'd', 'i', 's', 't', 'r', 'i', 'b', 'u', 't', 'e', 'd'
    };

    active_request =
        request_id_generator.next();

    if (!active_request.valid()) {
        Serial.println(
            "REQUEST ID GENERATION: FAIL"
        );

        request_started = true;
        return;
    }

    if (
        !build_set_request(
            active_request,
            test_key,
            test_key_length,
            value,
            sizeof(value),
            active_packet,
            sizeof(active_packet),
            active_packet_length
        )
    ) {
        Serial.println(
            "SET_REQ BUILD: FAIL"
        );

        request_started = true;
        return;
    }

    bool initial_send_ok = false;

    const request::Result submit_result =
        request_manager.submit(
            active_request,
            node::A,
            active_packet,
            active_packet_length,
            static_cast<std::uint64_t>(millis()),
            reliable_send,
            &sender_context,
            initial_send_ok
        );

    request_started = true;

    if (

namespace membership {

Manager::Manager(
    node::Id local_node_id,
    std::uint64_t timeout_ms
)
    : local_node_id_(local_node_id),
      timeout_ms_(timeout_ms)
{
}

bool Manager::valid_node(node::Id id)
{
    return id >= node::MIN &&
           id <= node::MAX;
}

std::size_t Manager::index(node::Id id)
{
    return static_cast<std::size_t>(
        id - node::MIN
    );
}

bool Manager::begin(std::uint64_t now_ms)
{
    if (!valid_node(local_node_id_)) {
        return false;
    }

    if (timeout_ms_ == 0) {
        return false;
    }

    for (std::size_t i = 0; i < NODE_COUNT; ++i) {
        neighbors_[i] = NeighborState{
            0,
            false
        };
    }

    /*
     * A node considers itself alive without requiring
     * a heartbeat from itself.
     */
    neighbors_[index(local_node_id_)] = NeighborState{
        now_ms,
        true
    };

    initialized_ = true;

    return true;
}

bool Manager::heartbeat_received(
    node::Id sender,
    std::uint64_t now_ms
)
{
    if (!initialized_ ||
        !valid_node(sender)) {

        return false;
    }

    /*
     * Heartbeats from ourselves are not meaningful
     * membership observations.
     */
    if (sender == local_node_id_) {
        return true;
    }

    const std::size_t i = index(sender);

    neighbors_[i].last_heard_ms = now_ms;
    neighbors_[i].seen = true;

    return true;
}

bool Manager::is_alive(
    node::Id id,
    std::uint64_t now_ms
) const
{
    if (!initialized_ ||
        !valid_node(id)) {

        return false;
    }

    if (id == local_node_id_) {
        return true;
    }

    const NeighborState& state =
        neighbors_[index(id)];

    if (!state.seen) {
        return false;
    }

    /*
     * Monotonic time means now_ms should never be
     * earlier than last_heard_ms.
     *
     * Treat an anomalous backwards clock reading
     * as not alive rather than underflowing.
     */
    if (now_ms < state.last_heard_ms) {
        return false;
    }

    const std::uint64_t elapsed =
        now_ms - state.last_heard_ms;

    return elapsed <= timeout_ms_;
}

std::uint64_t Manager::last_heard(
    node::Id id
) const
{
    if (!initialized_ ||
        !valid_node(id)) {

        return 0;
    }

    return neighbors_[index(id)].last_heard_ms;
}

bool Manager::seen(node::Id id) const
{
    if (!initialized_ ||
        !valid_node(id)) {

        return false;
    }

    return neighbors_[index(id)].seen;
}

node::Id Manager::local_node_id() const
{
    return local_node_id_;
}

std::uint64_t Manager::timeout_ms() const
{
    return timeout_ms_;
}

bool Manager::ready() const
{
    return initialized_;
}

bool Resolver::begin()

    }

    initialized_ = true;

    return true;
}

bool Resolver::ready() const

bool Resolver::locate(
    const std::uint8_t* key,
    std::size_t key_length,
    const membership::Manager& membership,
    std::uint64_t now_ms,
    Assignment& result
) const
{
    if (!initialized_) {
        return false;
    }

    if (key == nullptr && key_length != 0) {
        return false;
    }

    result = Assignment{
        node::INVALID,
        node::INVALID
    };

    const std::uint32_t key_hash =
        consistent_hash::fnv1a(
            key,
            key_length
        );

    std::size_t start_index = 0;

    if (
        !ring_.find_successor_index(
            key_hash,
            start_index
        )
    ) {
        return false;
    }

    /*
     * First live physical node clockwise
     * from the key position becomes primary.
     */
    for (
        std::size_t offset = 0;
        offset < ring_.point_count();
        ++offset
    ) {
        const std::size_t index =
            (start_index + offset) %
            ring_.point_count();

        const hash_ring::VirtualPoint& point =
            ring_.point(index);

        if (
            membership.is_alive(
                point.owner,
                now_ms
            )
        ) {
            result.primary = point.owner;
            break;
        }
    }

    /*
     * No live node exists.
     */
    if (result.primary == node::INVALID) {
        return false;
    }

    /*
     * Continue clockwise until we find a live node
     * belonging to a different physical machine.
     */
    for (
        std::size_t offset = 1;
        offset <= ring_.point_count();
        ++offset
    ) {
        const std::size_t index =
            (start_index + offset) %
            ring_.point_count();

        const hash_ring::VirtualPoint& point =
            ring_.point(index);

        if (
            point.owner != result.primary &&
            membership.is_alive(
                point.owner,
                now_ms
            )
        ) {
            result.replica = point.owner;
            break;
        }
    }

    return true;
}

namespace {

class PayloadWriter {
public:
    PayloadWriter(
        std::uint8_t* data,
        std::size_t capacity
    )
        : data_(data), capacity_(capacity), position_(0) {}

    bool write_u8(std::uint8_t value) {
        if (position_ >= capacity_) {
            return false;
        }

        data_[position_++] = value;
        return true;
    }

    bool write_bytes(
        const std::uint8_t* data,
        std::size_t len
    ) {
        if (len > capacity_ - position_) {
            return false;
        }

        if (len > 0) {
            std::memcpy(
                data_ + position_,
                data,
                len
            );
        }

        position_ += len;
        return true;
    }

    bool write_byte_view(ByteView view) {
        if (!write_u8(view.len)) {
            return false;
        }

        return write_bytes(view.data, view.len);
    }

    std::size_t size() const {
        return position_;
    }

private:
    std::uint8_t* data_;
    std::size_t capacity_;
    std::size_t position_;
};

class PayloadReader {
public:
    PayloadReader(
        const std::uint8_t* data,
        std::size_t length
    )
        : data_(data), length_(length), position_(0) {}

    bool read_u8(std::uint8_t& value) {
        if (position_ >= length_) {
            return false;
        }

        value = data_[position_++];
        return true;
    }

    bool read_byte_view(ByteView& view) {
        std::uint8_t len = 0;

        if (!read_u8(len)) {
            return false;
        }

        if (len > length_ - position_) {
            return false;
        }

        view = {
            data_ + position_,
            len
        };

        position_ += len;

        return true;
    }

    bool finished() const {
        return position_ == length_;
    }




private:
    const std::uint8_t* data_;
    std::size_t length_;
    std::size_t position_;
};

        payload.value.len > VALUE_MAX ||
        !reader.finished()) {
        return Result::INVALID_PAYLOAD;
    }

    payload.status =
        static_cast<StatusCode>(status);

    if (payload.status != StatusCode::OK &&
        payload.value.len != 0) {
        return Result::INVALID_PAYLOAD;
    }

    return Result::OK;
}

void write_u16_be(std::uint8_t* out, std::uint16_t value) {
    out[0] = static_cast<std::uint8_t>((value >> 8) & 0xFF);
    out[1] = static_cast<std::uint8_t>(value & 0xFF);
}

void write_u32_be(std::uint8_t* out, std::uint32_t value) {
    out[0] = static_cast<std::uint8_t>((value >> 24) & 0xFF);
    out[1] = static_cast<std::uint8_t>((value >> 16) & 0xFF);
    out[2] = static_cast<std::uint8_t>((value >> 8) & 0xFF);
    out[3] = static_cast<std::uint8_t>(value & 0xFF);
}

void write_u64_be(std::uint8_t* out, std::uint64_t value) {
    for (int i = 0; i < 8; ++i) {
        out[i] = static_cast<std::uint8_t>(
            (value >> (56 - 8 * i)) & 0xFF
        );
    }
}

std::uint16_t read_u16_be(const std::uint8_t* data) {
    return static_cast<std::uint16_t>(
        (static_cast<std::uint16_t>(data[0]) << 8) |
        static_cast<std::uint16_t>(data[1])
    );
}

std::uint32_t read_u32_be(const std::uint8_t* data) {
    return
        (static_cast<std::uint32_t>(data[0]) << 24) |
        (static_cast<std::uint32_t>(data[1]) << 16) |
        (static_cast<std::uint32_t>(data[2]) << 8) |
        static_cast<std::uint32_t>(data[3]);
}

std::uint64_t read_u64_be(const std::uint8_t* data) {
    std::uint64_t value = 0;

    for (int i = 0; i < 8; ++i) {
        value |= static_cast<std::uint64_t>(data[i])
                 << (56 - 8 * i);
    }

    return value;
}

Result serialize_version_table(
    const VersionTablePayload& payload,
    std::uint8_t* out,
    std::size_t out_capacity,
    std::size_t& bytes_written
) {
    bytes_written = 0;

    if (out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload.entry_count > VERSION_TABLE_MAX_ENTRIES) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.entry_count > 0 && payload.entries == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload.chunk_count == 0 ||
        payload.chunk_index >= payload.chunk_count) {
        return Result::INVALID_PAYLOAD;
    }

    std::size_t required = 9;

    for (std::size_t i = 0; i < payload.entry_count; ++i) {
        const VersionTableEntry& entry = payload.entries[i];

        if (entry.key.len > KEY_MAX) {
            return Result::INVALID_PAYLOAD;
        }

        if (entry.key.len > 0 &&
            entry.key.data == nullptr) {
            return Result::INVALID_ARGUMENT;
        }

        if (!is_valid_node_id(entry.version_origin_id)) {
            return Result::INVALID_PAYLOAD;
        }

        required +=
            static_cast<std::size_t>(11) +
            entry.key.len;
    }

    if (required > MAX_PAYLOAD_SIZE) {
        return Result::INVALID_PAYLOAD;
    }

    if (out_capacity < required) {
        return Result::BUFFER_TOO_SMALL;
    }

    write_u32_be(out, payload.sync_id);
    write_u16_be(out + 4, payload.chunk_index);
    write_u16_be(out + 6, payload.chunk_count);

    out[8] = payload.entry_count;

    std::size_t offset = 9;

    for (std::size_t i = 0; i < payload.entry_count; ++i) {
        const VersionTableEntry& entry = payload.entries[i];

        out[offset++] = entry.entry_flags;
        out[offset++] = entry.key.len;

        if (entry.key.len > 0) {
            std::memcpy(
                out + offset,
                entry.key.data,
                entry.key.len
            );

            offset += entry.key.len;
        }

        write_u64_be(out + offset, entry.version_hlc);
        offset += 8;

        out[offset++] = entry.version_origin_id;
    }

    bytes_written = offset;

    return Result::OK;
}


Result deserialize_version_table(
    const std::uint8_t* data,
    std::size_t payload_len,
    VersionTablePayload& payload,
    VersionTableEntry* entries_out,
    std::size_t entries_capacity
) {
    payload = {};

    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload_len < 9 ||
        payload_len > MAX_PAYLOAD_SIZE) {
        return Result::INVALID_PAYLOAD;
    }

    payload.sync_id = read_u32_be(data);
    payload.chunk_index = read_u16_be(data + 4);
    payload.chunk_count = read_u16_be(data + 6);
    payload.entry_count = data[8];

    if (payload.chunk_count == 0 ||
        payload.chunk_index >= payload.chunk_count) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.entry_count > VERSION_TABLE_MAX_ENTRIES) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.entry_count > entries_capacity) {
        return Result::BUFFER_TOO_SMALL;
    }

    if (payload.entry_count > 0 &&
        entries_out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    std::size_t offset = 9;

    for (std::size_t i = 0;
         i < payload.entry_count;
         ++i) {

        if (payload_len - offset < 11) {
            return Result::INVALID_PAYLOAD;
        }

        VersionTableEntry& entry = entries_out[i];

        entry.entry_flags = data[offset++];

        const std::uint8_t key_len = data[offset++];

        if (key_len > KEY_MAX) {
            return Result::INVALID_PAYLOAD;
        }

        if (payload_len - offset <
            static_cast<std::size_t>(key_len) + 9) {
            return Result::INVALID_PAYLOAD;
        }

        entry.key = {
            data + offset,
            key_len
        };

        offset += key_len;

        entry.version_hlc =
            read_u64_be(data + offset);

        offset += 8;

        entry.version_origin_id = data[offset++];

        if (!is_valid_node_id(entry.version_origin_id)) {
            return Result::INVALID_PAYLOAD;
        }
    }

    if (offset != payload_len) {
        return Result::INVALID_PAYLOAD;
    }

    payload.entries = entries_out;

    return Result::OK;
}

Result serialize_header(
    const PacketHeader& header,
    std::uint8_t* out,
    std::size_t out_capacity
) {
    if (out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (out_capacity < HEADER_SIZE) {
        return Result::BUFFER_TOO_SMALL;
    }

    if (header.protocol_version != PROTOCOL_VERSION ||
        header.header_len != HEADER_SIZE ||
        header.payload_len > MAX_PAYLOAD_SIZE ||
        header.ttl != 0 ||
        !is_valid_node_id(header.origin_node_id) ||
        !is_valid_node_id(header.sender_node_id) ||
        !is_valid_destination(header.destination_node_id) ||
        !is_known_message_type(header.msg_type) ||
        !has_valid_reserved_flags(header.flags)) {
        return Result::INVALID_PACKET;
    }

    out[0] = header.protocol_version;
    out[1] = static_cast<std::uint8_t>(header.msg_type);
    out[2] = header.flags;
    out[3] = header.header_len;

    write_u32_be(out + 4, header.request_id);

    out[8] = header.origin_node_id;
    out[9] = header.sender_node_id;
    out[10] = header.destination_node_id;
    out[11] = header.ttl;

    write_u64_be(out + 12, header.hlc);
    write_u16_be(out + 20, header.payload_len);

    return Result::OK;
}
Result deserialize_header(
    const std::uint8_t* data,
    std::size_t packet_len,
    PacketHeader& header
) {
    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (packet_len < HEADER_SIZE) {
        return Result::INVALID_PACKET;
    }

    header.protocol_version = data[0];
    header.msg_type =
        static_cast<MessageType>(data[1]);
    header.flags = data[2];
    header.header_len = data[3];

    header.request_id = read_u32_be(data + 4);

    header.origin_node_id = data[8];
    header.sender_node_id = data[9];
    header.destination_node_id = data[10];
    header.ttl = data[11];

    header.hlc = read_u64_be(data + 12);
    header.payload_len = read_u16_be(data + 20);

    if (header.protocol_version != PROTOCOL_VERSION ||
        header.header_len != HEADER_SIZE ||
        header.payload_len > MAX_PAYLOAD_SIZE ||
        packet_len != HEADER_SIZE + header.payload_len ||
        header.ttl != 0 ||
        !is_valid_node_id(header.origin_node_id) ||
        !is_valid_node_id(header.sender_node_id) ||
        !is_valid_destination(header.destination_node_id) ||
        !is_known_message_type(header.msg_type) ||
        !has_valid_reserved_flags(header.flags)) {
        return Result::INVALID_PACKET;
    }

    return Result::OK;
}

Result validate_message_header(
    const PacketHeader& header
) {
    switch (header.msg_type) {
        case MessageType::SET_REQ:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::SET_ACK:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::GET_REQ:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::GET_RESP:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::DEL_REQ:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::DEL_ACK:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::REPL_REQ:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::REPL_ACK:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::HEARTBEAT:
            if (header.destination_node_id != NODE_ID_BROADCAST) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::VERSION_TABLE:
            if (header.destination_node_id != NODE_ID_BROADCAST) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::RESYNC_DATA:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        default:
            return Result::INVALID_MESSAGE;
    }

    return Result::OK;
}
Result decode_message(
    const PacketView& packet,
    DecodedMessage& message
) {
    message = {};

    Result result = validate_message_header(packet.header);

    if (result != Result::OK) {
        return result;
    }

    message.header = packet.header;

    switch (packet.header.msg_type) {
        case MessageType::SET_REQ:
            return deserialize_set_request(
                packet.payload.data,
                packet.payload.len,
                message.payload.set_request
            );

        case MessageType::SET_ACK:
            return deserialize_set_ack(
                packet.payload.data,
                packet.payload.len,
                message.payload.set_ack
            );

        case MessageType::GET_REQ:
            return deserialize_get_request(
                packet.payload.data,
                packet.payload.len,
                message.payload.get_request
            );

        case MessageType::GET_RESP:
            return deserialize_get_response(
                packet.payload.data,
                packet.payload.len,
                message.payload.get_response
            );

        case MessageType::DEL_REQ:
            return deserialize_delete_request(
                packet.payload.data,
                packet.payload.len,
                message.payload.delete_request
            );

        case MessageType::DEL_ACK:
            return deserialize_delete_ack(
                packet.payload.data,
                packet.payload.len,
                message.payload.delete_ack
            );

        case MessageType::REPL_REQ:
            return deserialize_replication_request(
                packet.payload.data,
                packet.payload.len,
                message.payload.replication_request
            );

        case MessageType::REPL_ACK:
            return deserialize_replication_ack(
                packet.payload.data,
                packet.payload.len,
                message.payload.replication_ack
            );

        case MessageType::HEARTBEAT:
            if (packet.payload.len != 0) {
                return Result::INVALID_PAYLOAD;
            }

            return Result::OK;

        case MessageType::VERSION_TABLE:
            return deserialize_version_table(
                packet.payload.data,
                packet.payload.len,
                message.payload.version_table,
                message.version_table_entries,
                VERSION_TABLE_MAX_ENTRIES
            );

        case MessageType::RESYNC_DATA:
            return deserialize_resync_data(
                packet.payload.data,
                packet.payload.len,
                message.payload.resync_data
            );

        default:
            return Result::INVALID_MESSAGE;
    }
}



Result serialize_packet(
    const PacketHeader& header,
    ByteView payload,
    std::uint8_t* out,
    std::size_t out_capacity,
    std::size_t& bytes_written
) {
    bytes_written = 0;

    if (out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload.len > MAX_PAYLOAD_SIZE) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.len > 0 && payload.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (header.payload_len != payload.len) {
        return Result::INVALID_PACKET;
    }

    const std::size_t required =
        HEADER_SIZE + payload.len;

    if (out_capacity < required) {
        return Result::BUFFER_TOO_SMALL;
    }

    Result result = serialize_header(
        header,
        out,
        out_capacity
    );

    if (result != Result::OK) {
        return result;
    }

    if (payload.len > 0) {
        std::memcpy(
            out + HEADER_SIZE,
            payload.data,
            payload.len
        );
    }

    bytes_written = required;

    return Result::OK;
}

Result deserialize_packet(
    const std::uint8_t* data,
    std::size_t packet_len,
    PacketView& packet
) {
    packet = {};

    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    Result result = deserialize_header(
        data,
        packet_len,
        packet.header
    );

    if (result != Result::OK) {
        return result;
    }

    packet.payload = {
        data + HEADER_SIZE,
        static_cast<std::uint8_t>(packet.header.payload_len)
    };

    return Result::OK;
}

PacketResult extract_identity(

    identity = {};

    if (!is_reliable_message(

        )) {

        return PacketResult::NOT_RELIABLE;
    }

    if (packet.header.origin_node_id <
            node::MIN ||
        packet.header.origin_node_id >

    }

        return PacketResult::INVALID_PACKET;

PacketResult prepare_retry(
    const std::uint8_t* packet,
    std::size_t packet_length,
    std::uint8_t* retry_packet,
    std::size_t retry_capacity,
    std::size_t& retry_length
)
{
    retry_length = 0;

    if (packet == nullptr ||
        retry_packet == nullptr) {

        return PacketResult::INVALID_ARGUMENT;
    }

    protocol::PacketView view{};

    const protocol::Result decode_result =
        protocol::deserialize_packet(
            packet,
            packet_length,
            view
        );

    if (decode_result !=
        protocol::Result::OK) {

        return PacketResult::INVALID_PACKET;
    }

    if (!is_reliable_message(
            view.header.msg_type
        )) {

        return PacketResult::NOT_RELIABLE;
    }

    /*
     * A packet marked as RETRY is already a retry.
     * We do not need to manufacture another state.
     */
    view.header.flags |=
        protocol::FLAG_RETRY;

    /*
     * Re-serialize exactly the same header/payload,
     * except for the RETRY flag.
     */
    const protocol::Result serialize_result =
        protocol::serialize_packet(
            view.header,
            view.payload,
            retry_packet,
            retry_capacity,
            retry_length
        );

    if (serialize_result ==
        protocol::Result::BUFFER_TOO_SMALL) {

        return PacketResult::BUFFER_TOO_SMALL;
    }

    if (serialize_result !=
        protocol::Result::OK) {

        return PacketResult::INVALID_PACKET;
    }

    return PacketResult::OK;
}

bool same_request_content(
    const std::uint8_t* original_packet,
    std::size_t original_length,
    const std::uint8_t* retry_packet,
    std::size_t retry_length
)
{
    if (original_packet == nullptr ||
        retry_packet == nullptr) {

        return false;
    }

    protocol::PacketView original{};
    protocol::PacketView retry{};

    if (
        protocol::deserialize_packet(
            original_packet,
            original_length,
            original
        ) != protocol::Result::OK
    ) {
        return false;
    }

    if (
        protocol::deserialize_packet(
            retry_packet,
            retry_length,
            retry
        ) != protocol::Result::OK
    ) {
        return false;
    }

    /*
     * RETRY is the only allowed flag difference.
     */
    const std::uint8_t original_flags =
        original.header.flags;

    const std::uint8_t retry_flags =
        retry.header.flags;

    if (
        (retry_flags & ~protocol::FLAG_RETRY) !=
        original_flags
    ) {
        return false;
    }

    if (
        (retry_flags & protocol::FLAG_RETRY) == 0
    ) {
        return false;
    }

    /*
     * Everything except RETRY must remain identical.
     */
    if (
        original.header.protocol_version !=
            retry.header.protocol_version ||
        original.header.msg_type !=
            retry.header.msg_type ||
        original.header.header_len !=
            retry.header.header_len ||
        original.header.request_id !=
            retry.header.request_id ||
        original.header.origin_node_id !=
            retry.header.origin_node_id ||
        original.header.sender_node_id !=
            retry.header.sender_node_id ||
        original.header.destination_node_id !=
            retry.header.destination_node_id ||
        original.header.ttl !=
            retry.header.ttl ||
        original.header.hlc !=
            retry.header.hlc ||
        original.header.payload_len !=
            retry.header.payload_len
    ) {
        return false;
    }

    if (
        original.payload.len !=
        retry.payload.len
    ) {
        return false;
    }

    if (original.payload.len > 0) {

        if (original.payload.data == nullptr ||
            retry.payload.data == nullptr) {

            return false;
        }

        if (
            std::memcmp(
                original.payload.data,
                retry.payload.data,
                original.payload.len
            ) != 0
        ) {
            return false;
        }
    }

    return true;
}

bool Generator::begin(
    node::Id origin_node_id,
    std::uint32_t seed

        return false;
    }

    origin_node_id_ = origin_node_id;

    /*
     * seed represents the last already-reserved ID.
     *
     * next() increments before returning.
     *
     * Zero is reserved and will never be returned.
     */
    next_request_id_ = seed;

    if (next_request_id_ == 0) {
        next_request_id_ = 0;
    }

    initialized_ = true;

    return true;
}

Identity Generator::next()
{
    if (!initialized_) {
        return Identity{
            node::INVALID,
            0
        };
    }

    ++next_request_id_;

    /*
     * request_id = 0 is reserved.
     * Wrap directly to 1.
     */
    if (next_request_id_ == 0) {
        next_request_id_ = 1;
    }

    return Identity{
        origin_node_id_,
        next_request_id_
    };
}

bool Generator::ready() const
{
    return initialized_;
}

node::Id Generator::origin_node_id() const
{
    return origin_node_id_;
}

int Manager::find(
    Identity identity

{
    for (std::size_t i = 0;
         i < MAX_PENDING_REQUESTS;
         ++i) {

        if (entries_[i].state == State::FREE) {
            continue;
        }

        if (entries_[i].identity == identity) {
            return static_cast<int>(i);
        }
    }

    return -1;
}

    for (std::size_t i = 0;
         i < MAX_PENDING_REQUESTS;
         ++i) {

        if (entries_[i].state == State::FREE) {
            return static_cast<int>(i);
        }
    }

    return -1;
}

std::uint64_t Manager::retry_delay(
    std::uint64_t current_timeout
) const
{
    if (current_timeout >=
        config_.max_retry_timeout_ms) {

        return config_.max_retry_timeout_ms;
    }

    if (current_timeout >
        config_.max_retry_timeout_ms / 2) {

        return config_.max_retry_timeout_ms;
    }

    return current_timeout * 2;
}

void Manager::fail_entry(
    Entry& entry
)
{
    entry.state = State::FAILED;
}

Result Manager::submit(
    Identity identity,
    node::Id destination,
    const std::uint8_t* packet,
    std::size_t packet_length,
    std::uint64_t now_ms,
    SendFn send_fn,
    void* send_context,
    bool& initial_send_ok
)
{
    initial_send_ok = false;

    if (!initialized_) {
        return Result::NOT_READY;
    }

    if (!identity.valid()) {
        return Result::INVALID_ARGUMENT;
    }

    if (destination < node::MIN ||
        destination > node::MAX ||
        destination == identity.origin) {

        return Result::INVALID_ARGUMENT;
    }

    if (packet == nullptr ||
        packet_length == 0 ||
        packet_length > MAX_PACKET_SIZE) {

        return Result::INVALID_ARGUMENT;
    }

    if (send_fn == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (find(identity) >= 0) {
        return Result::DUPLICATE;
    }

    const int slot = find_free();

    if (slot < 0) {
        return Result::TABLE_FULL;
    }

    Entry& entry = entries_[slot];

    entry = Entry{};

    entry.identity = identity;
    entry.destination = destination;

    std::memcpy(
        entry.packet,
        packet,
        packet_length
    );

    entry.packet_length =
        static_cast<std::uint8_t>(
            packet_length
        );

    entry.created_at_ms = now_ms;

    entry.deadline_ms =
        now_ms +
        config_.overall_deadline_ms;

    entry.current_retry_timeout_ms =
        config_.initial_timeout_ms;

    entry.next_retry_ms =
        now_ms +
        config_.initial_timeout_ms;

    entry.retry_count = 0;

    entry.state = State::PENDING;

    entry.send_fn = send_fn;
    entry.send_context = send_context;

    /*
     * The original transmission is not a retry.
     * If transport immediately rejects it, the logical
     * request remains pending and will retry normally.
     */
    initial_send_ok =
        send_fn(
            send_context,
            destination,
            entry.packet,
            packet_length,
            false
        );

    return Result::OK;
}

void Manager::tick(
    std::uint64_t now_ms
)
{
    if (!initialized_) {
        return;
    }

    for (std::size_t i = 0;
         i < MAX_PENDING_REQUESTS;
         ++i) {

        Entry& entry = entries_[i];

        if (entry.state != State::PENDING) {
            continue;
        }

        /*
         * Overall deadline always wins.
         */
        if (now_ms >= entry.deadline_ms) {
            fail_entry(entry);
            continue;
        }

        if (now_ms < entry.next_retry_ms) {
            continue;
        }

        /*
         * Maximum retransmissions exhausted.
         */
        if (entry.retry_count >=
            config_.max_retransmissions) {

            /*
             * Leave the request pending until the overall
             * deadline so the terminal timeout remains
             * exactly the configured deadline.
             */
            entry.next_retry_ms =
                entry.deadline_ms;

            continue;
        }

        ++entry.retry_count;

        const bool send_ok =
            entry.send_fn(
                entry.send_context,
                entry.destination,
                entry.packet,
                entry.packet_length,
                true
            );

        (void)send_ok;

        /*
         * Backoff applies after the retry just transmitted.
         */
        entry.current_retry_timeout_ms =
            retry_delay(
                entry.current_retry_timeout_ms
            );

        entry.next_retry_ms =
            now_ms +
            entry.current_retry_timeout_ms;

        /*
         * Never schedule beyond the terminal deadline.
         */
        if (entry.next_retry_ms >
            entry.deadline_ms) {

            entry.next_retry_ms =
                entry.deadline_ms;
        }
    }
}

Result Manager::complete(
    Identity identity,
    const std::uint8_t* response,
    std::size_t response_length
)
{
    if (!initialized_) {
        return Result::NOT_READY;
    }

    if (!identity.valid()) {
        return Result::INVALID_ARGUMENT;
    }

    const int slot = find(identity);

    if (slot < 0) {
        /*
         * No matching request exists.
         * Treat it as a late/unknown response.
         */
        return Result::UNKNOWN_REQUEST;
    }

    Entry& entry = entries_[slot];

    if (entry.state != State::PENDING) {
        /*
         * Terminal request:
         * late duplicate response is ignored.
         */
        return Result::ALREADY_TERMINAL;
    }

    if (response_length > MAX_PACKET_SIZE) {
        return Result::BUFFER_TOO_SMALL;
    }

    if (response_length > 0 &&
        response == nullptr) {

        return Result::INVALID_ARGUMENT;
    }

    if (response_length > 0) {
        std::memcpy(
            entry.response,
            response,
            response_length
        );
    }

    entry.response_length =
        static_cast<std::uint8_t>(
            response_length
        );

    entry.state = State::COMPLETED;

    return Result::OK;
}

Result Manager::get_state(
    Identity identity,
    State& state
) const
{
    state = State::FREE;

    if (!initialized_) {
        return Result::NOT_READY;
    }

    const int slot = find(identity);

    if (slot < 0) {
        return Result::UNKNOWN_REQUEST;
    }

    state = entries_[slot].state;

    return Result::OK;
}

Result Manager::get_info(

    }

    }

    const Entry& entry = entries_[slot];

Result Manager::get_completion(
    Identity identity,
    std::uint8_t* output,
    std::size_t output_capacity,
    std::size_t& output_length
) const
{
    output_length = 0;

    if (!initialized_) {
        return Result::NOT_READY;
    }

    if (output == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    const int slot = find(identity);

    if (slot < 0) {
        return Result::UNKNOWN_REQUEST;
    }

    const Entry& entry = entries_[slot];

    if (entry.state != State::COMPLETED) {
        return Result::ALREADY_TERMINAL;
    }

    if (output_capacity <
        entry.response_length) {

        return Result::BUFFER_TOO_SMALL;
    }

    if (entry.response_length > 0) {
        std::memcpy(
            output,
            entry.response,
            entry.response_length
        );
    }

    output_length = entry.response_length;

    return Result::OK;
}

    }

        return Result::UNKNOWN_REQUEST;

    return Result::OK;
}

int compare_versions(
    const Version& lhs,
    const Version& rhs
) {
    if (lhs.hlc < rhs.hlc) {
        return -1;
    }

    if (lhs.hlc > rhs.hlc) {
        return 1;
    }

    if (lhs.origin < rhs.origin) {

    if (lhs.origin > rhs.origin) {

    }

    return 0;

Result StorageManager::map_eeprom_result(

) const
{
    switch (result) {
    case eeprom::Result::OK:
        return Result::OK;

    case eeprom::Result::INVALID_ARGUMENT:
        return Result::INVALID_ARGUMENT;

    case eeprom::Result::OUT_OF_RANGE:
        return Result::STORAGE_ERROR;

    case eeprom::Result::NOT_INITIALIZED:
        return Result::NOT_INITIALIZED;

    case eeprom::Result::I2C_ERROR:
    case eeprom::Result::WRITE_TIMEOUT:
        return Result::STORAGE_ERROR;

Result StorageManager::write_committed_block(
    uint16_t address,
    const uint8_t* data,
    size_t length,
    size_t commit_offset
)
{
    if (data == nullptr ||
        length == 0 ||
        length > metadata::SUPERBLOCK_SIZE ||
        commit_offset >= length) {
        return Result::INVALID_ARGUMENT;
    }

    uint8_t temporary[metadata::SUPERBLOCK_SIZE]{};

    std::memcpy(
        temporary,
        data,
        length
    );

    const uint8_t commit_marker =
        temporary[commit_offset];

    // Suppress the commit marker during the first write.
    temporary[commit_offset] = 0x00;

    eeprom::Result eeprom_result =
        eeprom_.write(
            address,
            temporary,
            length
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    // Verify everything written before commit.
    uint8_t verify[metadata::SUPERBLOCK_SIZE]{};

    eeprom_result =
        eeprom_.read(
            address,
            verify,
            length
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    if (std::memcmp(
            verify,
            temporary,
            length
        ) != 0) {
        return Result::STORAGE_ERROR;
    }

    // Commit marker is deliberately written last.
    eeprom_result =
        eeprom_.write(
            static_cast<uint16_t>(
                address + commit_offset
            ),
            &commit_marker,
            1
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    // Verify the complete committed block.
    eeprom_result =
        eeprom_.read(
            address,
            verify,
            length
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    if (std::memcmp(
            verify,
            data,
            length
        ) != 0) {
        return Result::STORAGE_ERROR;
    }

    return Result::OK;
}

Result StorageManager::clear_range(
    uint16_t address,
    size_t length
)
{
    constexpr size_t CHUNK_SIZE = 64;

    uint8_t erased[CHUNK_SIZE];

    for (size_t i = 0; i < CHUNK_SIZE; ++i) {
        erased[i] = 0xFF;
    }

    size_t remaining = length;
    uint16_t current = address;

    while (remaining > 0) {
        const size_t chunk =
            remaining > CHUNK_SIZE
                ? CHUNK_SIZE
                : remaining;

        const eeprom::Result result =
            eeprom_.write(
                current,
                erased,
                chunk
            );

        if (result != eeprom::Result::OK) {
            return map_eeprom_result(result);
        }

        current = static_cast<uint16_t>(
            current + chunk
        );

        remaining -= chunk;
    }

    return Result::OK;
}


Result StorageManager::initialize_empty_store()
{
    // Use Bank A as the first authoritative bank.
    active_bank_ = metadata::Bank::A;
    superblock_generation_ = 1;
    active_bank_generation_ = 1;
    next_sequence_ = 1;

    append_address_ =
        static_cast<uint16_t>(
            metadata::BANK_A_BASE +
            metadata::BANK_RECORD_OFFSET
        );

    record_count_ = 0;

    // Clear Bank A record area so every byte after the
    // append position is known to be unused.
    Result result = clear_range(
        static_cast<uint16_t>(
            metadata::BANK_A_BASE +
            metadata::BANK_RECORD_OFFSET
        ),
        metadata::BANK_RECORD_AREA_END -
        metadata::BANK_RECORD_OFFSET
    );

    if (result != Result::OK) {
        return result;
    }

    // Create the initial active bank as OPEN.
    result = write_bank_header(
        active_bank_,
        metadata::BankState::OPEN,
        active_bank_generation_,
        0,
        metadata::BANK_RECORD_OFFSET
    );

    if (result != Result::OK) {
        return result;
    }

    // Create an empty footer checkpoint.
    result = write_bank_footer(
        active_bank_,
        active_bank_generation_,
        0,
        metadata::BANK_RECORD_OFFSET
    );

    if (result != Result::OK) {
        return result;
    }

    // Finally publish Bank A through Superblock generation 1.
    //
    // This is the point at which the empty store becomes
    // authoritative for recovery.
    result = write_superblock(
        superblock_generation_,
        active_bank_,
        active_bank_generation_,
        0,
        0
    );

    if (result != Result::OK) {
        return result;
    }

    return Result::OK;
}

bool StorageManager::erased_tail(
    uint16_t address
)
{
    constexpr size_t CHUNK_SIZE = 64;

    const uint16_t end_address =
        static_cast<uint16_t>(
            bank_header_address(active_bank_) +
            metadata::BANK_RECORD_AREA_END
        );

    uint8_t buffer[CHUNK_SIZE]{};

    uint16_t current = address;

    while (current < end_address) {
        const size_t remaining =
            static_cast<size_t>(
                end_address - current
            );

        const size_t chunk =
            remaining > CHUNK_SIZE
                ? CHUNK_SIZE
                : remaining;

        const eeprom::Result result =
            eeprom_.read(
                current,
                buffer,
                chunk
            );

        if (result != eeprom::Result::OK) {
            return false;
        }

        for (size_t i = 0; i < chunk; ++i) {
            if (buffer[i] != 0xFF) {
                return false;
            }
        }

        current = static_cast<uint16_t>(
            current + chunk
        );
    }

    return true;
}

Result StorageManager::recover_active_bank()
{
    // RAM index is rebuilt entirely from persistent records.
    std::memset(
        index_,
        0,
        sizeof(index_)
    );

    const uint16_t record_start =
        bank_record_start(active_bank_);

    const uint16_t record_end =
        static_cast<uint16_t>(
            bank_header_address(active_bank_) +
            metadata::BANK_RECORD_AREA_END
        );

    uint16_t current_address = record_start;

    uint32_t highest_sequence = 0;
    uint16_t committed_records = 0;

    while (current_address < record_end) {

        // First check whether the next record area is erased.
        uint8_t probe[2]{};

        eeprom::Result eeprom_result =
            eeprom_.read(
                current_address,
                probe,
                sizeof(probe)
            );

        if (eeprom_result != eeprom::Result::OK) {
            return map_eeprom_result(eeprom_result);
        }

        // Known unused tail.
        if (probe[0] == 0xFF &&
            probe[1] == 0xFF) {
            break;
        }

        // Read the fixed record header.
        uint8_t header[
            record::HEADER_SIZE
        ]{};

        eeprom_result =
            eeprom_.read(
                current_address,
                header,
                sizeof(header)
            );

        if (eeprom_result != eeprom::Result::OK) {
            return map_eeprom_result(eeprom_result);
        }

        const uint16_t record_length =
            static_cast<uint16_t>(
                (static_cast<uint16_t>(header[4]) << 8) |
                static_cast<uint16_t>(header[5])
            );

        const bool plausible_header =
            (header[0] == 0x52) &&
            (header[1] == 0x43) &&
            record_length >=
                record::HEADER_SIZE +
                record::CRC_SIZE +
                record::COMMIT_MARKER_SIZE &&
            record_length <=
                record::MAX_RECORD_SIZE &&
            record_length <=
                static_cast<uint16_t>(
                    record_end - current_address
                );

        if (!plausible_header) {
            // If everything after this position is erased,
            // this is an incomplete tail record.
            if (erased_tail(current_address)) {
                break;
            }

            return Result::CORRUPTED;
        }

        uint8_t record_buffer[
            record::MAX_RECORD_SIZE
        ]{};

        eeprom_result =
            eeprom_.read(
                current_address,
                record_buffer,
                record_length
            );

        if (eeprom_result != eeprom::Result::OK) {
            return map_eeprom_result(eeprom_result);
        }

        record::Record persistent_record{};

        const Result decode_result =
            record::deserialize(
                record_buffer,
                record_length,
                persistent_record
            );

        if (decode_result != Result::OK) {
            // Same tail-vs-middle distinction.
            if (erased_tail(current_address)) {
                break;
            }

            return Result::CORRUPTED;
        }

        // Track physical sequence information.
        if (persistent_record.sequence >
            highest_sequence) {
            highest_sequence =
                persistent_record.sequence;
        }

        ++committed_records;

        // Rebuild logical state.
        const int existing_index =
            find_index_entry(
                persistent_record.key
            );

        if (existing_index >= 0) {

            IndexEntry& current_entry =
                index_[existing_index];

            const Version current_version{
                current_entry.version_hlc,
                current_entry.version_origin
            };

            if (compare_versions(
                    persistent_record.version,
                    current_version
                ) > 0) {

                update_index_entry(
                    static_cast<size_t>(
                        existing_index
                    ),
                    persistent_record.key,
                    persistent_record.state,
                    static_cast<uint16_t>(
                        persistent_record.value.len
                    ),
                    current_address,
                    persistent_record.version
                );
            }

        } else {

            const int free_index =
                find_free_index_entry();

            if (free_index < 0) {
                return Result::INDEX_FULL;
            }

            update_index_entry(
                static_cast<size_t>(free_index),
                persistent_record.key,
                persistent_record.state,
                static_cast<uint16_t>(
                    persistent_record.value.len
                ),
                current_address,
                persistent_record.version
            );
        }

        current_address =
            static_cast<uint16_t>(
                current_address + record_length
            );
    }

    append_address_ = current_address;

    record_count_ = committed_records;

    next_sequence_ =
        highest_sequence == 0
            ? 1
            : highest_sequence + 1;

    return Result::OK;
}

Result StorageManager::append_record(
    KeyView key,
    ValueView value,
    Version version,
    State state,
    uint16_t& record_address
)
{
    return append_record_to_bank(
        active_bank_,
        append_address_,
        next_sequence_,
        key,
        value,
        version,
        state,
        record_address
    );
}
Result StorageManager::append_record_to_bank(
    metadata::Bank bank,
    uint16_t& append_address,
    uint32_t& next_sequence,
    KeyView key,
    ValueView value,
    Version version,
    State state,
    uint16_t& record_address
)
{
    if (!initialized_) {
        return Result::NOT_INITIALIZED;
    }

    if (key.len == 0) {
        return Result::INVALID_ARGUMENT;
    }

    if (key.len > KEY_MAX) {
        return Result::KEY_TOO_LARGE;
    }

    if (key.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (value.len > VALUE_MAX) {
        return Result::VALUE_TOO_LARGE;
    }

    if (value.len > 0 && value.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (state != State::LIVE &&
        state != State::TOMBSTONE) {
        return Result::INVALID_ARGUMENT;
    }

    if (state == State::TOMBSTONE &&
        value.len != 0) {
        return Result::INVALID_ARGUMENT;
    }

    if (!metadata::is_valid_bank(bank)) {
        return Result::INVALID_ARGUMENT;
    }

    record::Record persistent_record{
        state,
        version,
        next_sequence,
        key,
        value
    };

    uint8_t buffer[record::MAX_RECORD_SIZE]{};
    size_t record_length = 0;

    Result result = record::serialize(
        persistent_record,
        buffer,
        sizeof(buffer),
        record_length
    );

    if (result != Result::OK) {
        return result;
    }

    const uint16_t record_area_end =
        static_cast<uint16_t>(
            bank_header_address(bank) +
            metadata::BANK_RECORD_AREA_END
        );

    if (append_address < bank_record_start(bank) ||
        append_address > record_area_end) {
        return Result::CORRUPTED;
    }

    const size_t remaining =
        static_cast<size_t>(
            record_area_end - append_address
        );

    if (record_length > remaining) {
        return Result::NO_SPACE;
    }

    record_address = append_address;

    // Write everything except the commit marker.
    eeprom::Result eeprom_result =
        eeprom_.write(
            record_address,
            buffer,
            record_length - 1
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    // Verify record body before committing.
    uint8_t verify[
        record::MAX_RECORD_SIZE
    ]{};

    eeprom_result =
        eeprom_.read(
            record_address,
            verify,
            record_length - 1
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    if (std::memcmp(
            verify,
            buffer,
            record_length - 1
        ) != 0) {
        return Result::STORAGE_ERROR;
    }

    // Commit marker is written last.
    const uint8_t commit_marker =
        buffer[record_length - 1];

    eeprom_result =
        eeprom_.write(
            static_cast<uint16_t>(
                record_address + record_length - 1
            ),
            &commit_marker,
            1
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    // Verify complete committed record.
    eeprom_result =
        eeprom_.read(
            record_address,
            verify,
            record_length
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    if (std::memcmp(
            verify,
            buffer,
            record_length
        ) != 0) {
        return Result::STORAGE_ERROR;
    }

    append_address =
        static_cast<uint16_t>(
            append_address + record_length
        );

    ++next_sequence;

    return Result::OK;
}

Result StorageManager::compact()
{
    if (!initialized_) {
        return Result::NOT_INITIALIZED;
    }

    const metadata::Bank source_bank =
        active_bank_;

    const metadata::Bank destination_bank =
        source_bank == metadata::Bank::A
            ? metadata::Bank::B
            : metadata::Bank::A;

    const uint32_t new_bank_generation =
        active_bank_generation_ + 1;

    const uint32_t new_superblock_generation =
        superblock_generation_ + 1;

    uint16_t destination_append_address =
        bank_record_start(destination_bank);

    uint32_t destination_next_sequence =
        next_sequence_;

    uint16_t new_record_count = 0;

    // Temporary record-address map.
    //
    // We must NOT modify the RAM index until the new bank
    // is completely committed. Otherwise a failed compaction
    // could leave the index pointing at an incomplete bank.
    uint16_t new_record_addresses[
        MAX_LOGICAL_KEYS
    ]{};

    std::memset(
        new_record_addresses,
        0,
        sizeof(new_record_addresses)
    );

    // --------------------------------------------------
    // 1. Clear destination bank record area.
    // --------------------------------------------------

    Result result = clear_range(
        bank_record_start(destination_bank),
        metadata::BANK_RECORD_AREA_END -
        metadata::BANK_RECORD_OFFSET
    );

    if (result != Result::OK) {
        return result;
    }

    // --------------------------------------------------
    // 2. Create destination bank as OPEN.
    // --------------------------------------------------

    result = write_bank_header(
        destination_bank,
        metadata::BankState::OPEN,
        new_bank_generation,
        0,
        metadata::BANK_RECORD_OFFSET
    );

    if (result != Result::OK) {
        return result;
    }

    // --------------------------------------------------
    // 3. Copy exactly one current record per logical key.
    //
    // This includes TOMBSTONE entries.
    // --------------------------------------------------

    for (std::size_t i = 0;
         i < MAX_LOGICAL_KEYS;
         ++i) {

        const IndexEntry& entry =
            index_[i];

        if (!entry.valid) {
            continue;
        }

        if (entry.state != State::LIVE &&
            entry.state != State::TOMBSTONE) {
            return Result::CORRUPTED;
        }

        uint8_t value_buffer[
            VALUE_MAX
        ]{};

        size_t value_length = 0;

        if (entry.state == State::LIVE) {

            result = read_value(
                entry,
                value_buffer,
                sizeof(value_buffer),
                value_length
            );

            if (result != Result::OK) {
                return result;
            }

        } else {

            value_length = 0;
        }

        const Version version{
            entry.version_hlc,
            entry.version_origin
        };

        uint16_t destination_record_address = 0;

        result = append_record_to_bank(
            destination_bank,
            destination_append_address,
            destination_next_sequence,
            KeyView{
                entry.key,
                entry.key_len
            },
            ValueView{
                entry.state == State::LIVE
                    ? value_buffer
                    : nullptr,
                value_length
            },
            version,
            entry.state,
            destination_record_address
        );

        if (result != Result::OK) {
            return result;
        }

        new_record_addresses[i] =
            destination_record_address;

        ++new_record_count;
    }

    // --------------------------------------------------
    // 4. Write footer for the completed destination bank.
    // --------------------------------------------------

    const uint16_t destination_records_end =
        static_cast<uint16_t>(
            destination_append_address -
            bank_header_address(destination_bank)
        );

    result = write_bank_footer(
        destination_bank,
        new_bank_generation,
        new_record_count,
        destination_records_end
    );

    if (result != Result::OK) {
        return result;
    }

    // --------------------------------------------------
    // 5. Seal destination bank.
    // --------------------------------------------------

    result = write_bank_header(
        destination_bank,
        metadata::BankState::SEALED,
        new_bank_generation,
        new_record_count,
        destination_records_end
    );

    if (result != Result::OK) {
        return result;
    }

    // --------------------------------------------------
    // 6. Publish the new bank through the new superblock.
    //
    // This is the atomic authority switch.
    // --------------------------------------------------

    const uint32_t highest_sequence =
    destination_next_sequence == 0
        ? 0
        : destination_next_sequence - 1;

    result = write_superblock(
        new_superblock_generation,
        destination_bank,
        new_bank_generation,
        highest_sequence,
        new_record_count
    );

    if (result != Result::OK) {
        // Old bank remains authoritative because the new
        // superblock was never committed.
        return result;
    }
    

    // --------------------------------------------------
    // 7. Only NOW update runtime state.
    // --------------------------------------------------

    for (std::size_t i = 0;
         i < MAX_LOGICAL_KEYS;
         ++i) {

        if (!index_[i].valid) {
            continue;
        }

        index_[i].record_address =
            new_record_addresses[i];
    }

    active_bank_ =
        destination_bank;

    superblock_generation_ =
        new_superblock_generation;

    active_bank_generation_ =
        new_bank_generation;

    append_address_ =
        destination_append_address;

    next_sequence_ =
        destination_next_sequence;

    record_count_ =
        new_record_count;

    return Result::OK;
}

Result StorageManager::apply(
    KeyView key,
    ValueView value,
    Version version,
    State state
)
{
    if (!initialized_) {
        return Result::NOT_INITIALIZED;
    }

    if (key.len == 0) {
        return Result::INVALID_ARGUMENT;
    }

    if (key.len > KEY_MAX) {
        return Result::KEY_TOO_LARGE;
    }

    if (key.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (value.len > VALUE_MAX) {
        return Result::VALUE_TOO_LARGE;
    }

    if (value.len > 0 && value.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (state != State::LIVE &&
        state != State::TOMBSTONE) {
        return Result::INVALID_ARGUMENT;
    }

    if (state == State::TOMBSTONE &&
        value.len != 0) {
        return Result::INVALID_ARGUMENT;
    }

    const int existing_index =
        find_index_entry(key);

    int target_index = existing_index;

    if (existing_index >= 0) {

        const IndexEntry& entry =
            index_[existing_index];

        const Version current_version{
            entry.version_hlc,
            entry.version_origin
        };

        const int version_comparison =
            compare_versions(
                version,
                current_version
            );

        if (version_comparison <= 0) {
            return Result::STALE_VERSION;
        }

    } else {

        target_index = find_free_index_entry();

        if (target_index < 0) {
            return Result::INDEX_FULL;
        }
    }

    uint16_t record_address = 0;

    const Result append_result =
        append_record(
            key,
            value,
            version,
            state,
            record_address
        );

    if (append_result != Result::OK) {
        return append_result;
    }

    update_index_entry(
        static_cast<std::size_t>(target_index),
        key,
        state,
        static_cast<uint16_t>(value.len),
        record_address,
        version
    );

    return Result::OK;
}
Result StorageManager::set(
    KeyView key,
    ValueView value,
    Version version
)
{
    return apply(
        key,
        value,
        version,
        State::LIVE
    );
}

Result StorageManager::remove(
    KeyView key,
    Version version
)
{
    return apply(
        key,
        ValueView{nullptr, 0},
        version,
        State::TOMBSTONE
    );
}

Result StorageManager::read_value(
    const IndexEntry& entry,
    uint8_t* value_out,
    size_t value_capacity,
    size_t& value_len
)
{
    value_len = 0;

    if (!initialized_) {
        return Result::NOT_INITIALIZED;
    }

    if (entry.state == State::TOMBSTONE) {
        return Result::NOT_FOUND;
    }

    if (entry.value_len > value_capacity) {
        return Result::BUFFER_TOO_SMALL;
    }

    if (entry.value_len > 0 && value_out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    uint8_t buffer[record::MAX_RECORD_SIZE]{};

    uint8_t header[record::HEADER_SIZE]{};

    eeprom::Result eeprom_result =
        eeprom_.read(
            entry.record_address,
            header,
            sizeof(header)
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    const uint16_t record_length =
        static_cast<uint16_t>(
            (static_cast<uint16_t>(header[4]) << 8) |
            static_cast<uint16_t>(header[5])
        );

    if (record_length < record::HEADER_SIZE +
                           record::CRC_SIZE +
                           record::COMMIT_MARKER_SIZE ||
        record_length > record::MAX_RECORD_SIZE) {
        return Result::CORRUPTED;
    }

    eeprom_result =
        eeprom_.read(
            entry.record_address,
            buffer,
            record_length
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    record::Record persistent_record{};

    const Result decode_result =
        record::deserialize(
            buffer,
            record_length,
            persistent_record
        );

    if (decode_result != Result::OK) {
        return decode_result;
    }

    if (persistent_record.state != State::LIVE ||
        persistent_record.key.len != entry.key_len ||
        std::memcmp(
            persistent_record.key.data,
            entry.key,
            entry.key_len
        ) != 0 ||
        persistent_record.version.hlc != entry.version_hlc ||
        persistent_record.version.origin != entry.version_origin) {
        return Result::CORRUPTED;
    }

    if (persistent_record.value.len > 0) {
        std::memcpy(
            value_out,
            persistent_record.value.data,
            persistent_record.value.len
        );
    }

    value_len = persistent_record.value.len;

    return Result::OK;
}

Result StorageManager::get(
    KeyView key,
    uint8_t* value_out,
    size_t value_capacity,
    size_t& value_len
)
{
    value_len = 0;

    if (!initialized_) {
        return Result::NOT_INITIALIZED;
    }

    if (key.len == 0 ||
        key.len > KEY_MAX ||
        (key.len > 0 && key.data == nullptr)) {
        return key.len > KEY_MAX
            ? Result::KEY_TOO_LARGE
            : Result::INVALID_ARGUMENT;
    }

    const int index =
        find_index_entry(key);

    if (index < 0) {
        return Result::NOT_FOUND;
    }

    const IndexEntry& entry =
        index_[index];

    if (entry.state == State::TOMBSTONE) {
        return Result::NOT_FOUND;
    }

    return read_value(
        entry,
        value_out,
        value_capacity,
        value_len
    );
}



Result StorageManager::begin()
{
    if (initialized_) {
        return Result::OK;
    }

    eeprom::Result eeprom_result =
        eeprom_.begin(
            EEPROM_SDA_PIN,
            EEPROM_SCL_PIN
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    uint8_t slot0[
        metadata::SUPERBLOCK_SIZE
    ]{};

    uint8_t slot1[
        metadata::SUPERBLOCK_SIZE
    ]{};

    eeprom_result =
        eeprom_.read(
            metadata::SUPERBLOCK_SLOT_0,
            slot0,
            sizeof(slot0)
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    eeprom_result =
        eeprom_.read(
            metadata::SUPERBLOCK_SLOT_1,
            slot1,
            sizeof(slot1)
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    metadata::Superblock sb0{};
    metadata::Superblock sb1{};

    const bool valid0 =
        deserialize_superblock(
            slot0,
            sizeof(slot0),
            sb0
        ) == Result::OK;

    const bool valid1 =
        deserialize_superblock(
            slot1,
            sizeof(slot1),
            sb1
        ) == Result::OK;

    // Fresh EEPROM.
    if (!valid0 && !valid1) {
        const Result result =
            initialize_empty_store();
            Serial.println("Storage: no valid superblock, initializing fresh store");
        if (result != Result::OK) {
            return result;
        }

        initialized_ = true;
        return Result::OK;
    }

    // Select newest valid superblock.
    metadata::Superblock selected{};

    if (valid0 && valid1) {
        selected =
            sb0.generation >= sb1.generation
                ? sb0
                : sb1;
    } else {
        selected = valid0 ? sb0 : sb1;
    }

    active_bank_ = selected.active_bank;
    Serial.println("Storage: valid superblock found, recovering existing store");
    superblock_generation_ = selected.generation;
    active_bank_generation_ =
        selected.active_bank_generation;
    next_sequence_ =
        selected.last_known_sequence + 1;
    record_count_ =
        selected.stored_record_count;

    // For this checkpoint we only support
    // reopening an empty persistent bank.
    uint8_t header_buffer[
        metadata::BANK_HEADER_SIZE
    ]{};

    eeprom_result =
        eeprom_.read(
            bank_header_address(active_bank_),
            header_buffer,
            sizeof(header_buffer)
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    metadata::BankHeader header{};

    if (deserialize_bank_header(
            header_buffer,
            sizeof(header_buffer),
            header
        ) != Result::OK) {
        return Result::CORRUPTED;
    }

    uint8_t footer_buffer[
        metadata::BANK_FOOTER_SIZE
    ]{};

    eeprom_result =
        eeprom_.read(
            bank_footer_address(active_bank_),
            footer_buffer,
            sizeof(footer_buffer)
        );

    if (eeprom_result != eeprom::Result::OK) {
        return map_eeprom_result(eeprom_result);
    }

    metadata::BankFooter footer{};

    if (deserialize_bank_footer(
            footer_buffer,
            sizeof(footer_buffer),
            footer
        ) != Result::OK) {
        return Result::CORRUPTED;
    }

    if ((header.state != metadata::BankState::OPEN &&
     header.state != metadata::BankState::SEALED) ||
    header.generation != selected.active_bank_generation ||
    footer.generation != header.generation) {
    return Result::CORRUPTED;
}
        const Result recovery_result =
        recover_active_bank();

    if (recovery_result != Result::OK) {
        return recovery_result;
    }

    

    initialized_ = true;
    return Result::OK;
}

bool StorageManager::ready() const
{
    return initialized_;
}

Result StorageManager::max_persisted_hlc(Hlc& out) const
{
    out = 0;

    if (!initialized_) {
        return Result::NOT_INITIALIZED;
    }

    for (std::size_t i = 0; i < MAX_LOGICAL_KEYS; ++i) {
        const IndexEntry& entry = index_[i];

        if (!entry.valid) {
            continue;
        }

        if (entry.version_hlc > out) {
            out = entry.version_hlc;
        }
    }

    return Result::OK;
}

constexpr size_t SUPERBLOCK_COMMIT_OFFSET = 23;

void write_u16_be(uint8_t* data, size_t offset, uint16_t value)
{
    data[offset] =
        static_cast<uint8_t>((value >> 8) & 0xFFu);
    data[offset + 1] =
        static_cast<uint8_t>(value & 0xFFu);
}

void write_u32_be(uint8_t* data, size_t offset, uint32_t value)
{
    data[offset] =
        static_cast<uint8_t>((value >> 24) & 0xFFu);
    data[offset + 1] =
        static_cast<uint8_t>((value >> 16) & 0xFFu);
    data[offset + 2] =
        static_cast<uint8_t>((value >> 8) & 0xFFu);
    data[offset + 3] =
        static_cast<uint8_t>(value & 0xFFu);
}

uint16_t read_u16_be(const uint8_t* data, size_t offset)
{
    return static_cast<uint16_t>(
        (static_cast<uint16_t>(data[offset]) << 8) |
        static_cast<uint16_t>(data[offset + 1])
    );
}

uint32_t read_u32_be(const uint8_t* data, size_t offset)
{
    return
        (static_cast<uint32_t>(data[offset]) << 24) |
        (static_cast<uint32_t>(data[offset + 1]) << 16) |
        (static_cast<uint32_t>(data[offset + 2]) << 8) |
        static_cast<uint32_t>(data[offset + 3]);
}

bool reserved_zero(
    const uint8_t* data,
    size_t offset,
    size_t length
)
{
    for (size_t i = 0; i < length; ++i) {
        if (data[offset + i] != 0) {
            return false;
        }
    }

    return true;
}

Result serialize_superblock(

    uint8_t* output,
    size_t output_capacity
)
{
    if (output == nullptr ||
        output_capacity < SUPERBLOCK_SIZE) {
        return Result::BUFFER_TOO_SMALL;
    }

    if (!is_valid_bank(value.active_bank)) {
        return Result::INVALID_ARGUMENT;
    }

    std::memset(output, 0, SUPERBLOCK_SIZE);

    write_u16_be(output, 0, SUPERBLOCK_MAGIC);

    output[2] = FORMAT_VERSION;
    output[3] = 0;

    write_u32_be(output, 4, value.generation);

    output[8] = static_cast<uint8_t>(value.active_bank);

    write_u32_be(
        output,
        9,
        value.active_bank_generation
    );

    write_u16_be(output, 13, 0);

    write_u32_be(
        output,
        15,
        value.last_known_sequence
    );

    write_u16_be(
        output,
        19,
        value.stored_record_count
    );

    const uint16_t crc =
        record::crc16(output, SUPERBLOCK_CRC_OFFSET);

    write_u16_be(output, SUPERBLOCK_CRC_OFFSET, crc);

    // Written last by the persistent layer in actual EEPROM operation.
    output[SUPERBLOCK_COMMIT_OFFSET] = COMMIT_MARKER;

    return Result::OK;
}

Result deserialize_superblock(
    const uint8_t* data,
    size_t length,
    Superblock& value
)
{
    value = Superblock{};

    if (data == nullptr || length != SUPERBLOCK_SIZE) {
        return Result::CORRUPTED;
    }

    if (read_u16_be(data, 0) != SUPERBLOCK_MAGIC) {
        return Result::CORRUPTED;
    }

    if (data[2] != FORMAT_VERSION) {
        return Result::CORRUPTED;
    }

    if (data[3] != 0 ||
        !reserved_zero(data, 24, 40)) {
        return Result::CORRUPTED;
    }

    if (!is_valid_bank(static_cast<Bank>(data[8]))) {
        return Result::CORRUPTED;
    }

    if (data[SUPERBLOCK_COMMIT_OFFSET] != COMMIT_MARKER) {
        return Result::CORRUPTED;
    }

    const uint16_t stored_crc =
        read_u16_be(data, SUPERBLOCK_CRC_OFFSET);

    const uint16_t calculated_crc =
        record::crc16(data, SUPERBLOCK_CRC_OFFSET);

    if (stored_crc != calculated_crc) {
        return Result::CORRUPTED;
    }

    value.generation = read_u32_be(data, 4);

    value.active_bank =
        static_cast<Bank>(data[8]);

    value.active_bank_generation =
        read_u32_be(data, 9);

    value.last_known_sequence =
        read_u32_be(data, 15);

    value.stored_record_count =
        read_u16_be(data, 19);

    return Result::OK;
}

#include "storage_record.h"

namespace storage::record {

constexpr std::size_t OFFSET_MAGIC = 0;
constexpr std::size_t OFFSET_FORMAT_VERSION = 2;
constexpr std::size_t OFFSET_FLAGS = 3;
constexpr std::size_t OFFSET_RECORD_LEN = 4;
constexpr std::size_t OFFSET_KEY_LEN = 6;
constexpr std::size_t OFFSET_VALUE_LEN = 7;
constexpr std::size_t OFFSET_VERSION_HLC = 9;
constexpr std::size_t OFFSET_VERSION_ORIGIN = 17;
constexpr std::size_t OFFSET_SEQUENCE = 18;
constexpr std::size_t OFFSET_KEY = 22;

constexpr std::size_t MIN_RECORD_SIZE =
    HEADER_SIZE + CRC_SIZE + COMMIT_MARKER_SIZE;

constexpr std::uint8_t FLAGS_TOMBSTONE = 0x01;
constexpr std::uint8_t FLAGS_RESERVED_MASK = 0xFE;

void write_u16_be(
    std::uint8_t* out,
    std::uint16_t value
) {
    out[0] = static_cast<std::uint8_t>(value >> 8);
    out[1] = static_cast<std::uint8_t>(value & 0xFF);
}

void write_u32_be(
    std::uint8_t* out,
    std::uint32_t value
) {
    out[0] = static_cast<std::uint8_t>(value >> 24);
    out[1] = static_cast<std::uint8_t>(value >> 16);
    out[2] = static_cast<std::uint8_t>(value >> 8);
    out[3] = static_cast<std::uint8_t>(value & 0xFF);
}

void write_u64_be(
    std::uint8_t* out,
    std::uint64_t value
) {
    for (std::size_t i = 0; i < 8; ++i) {
        out[i] = static_cast<std::uint8_t>(
            value >> (56 - (i * 8))
        );
    }
}

std::uint16_t read_u16_be(
    const std::uint8_t* data
) {
    return static_cast<std::uint16_t>(
        (static_cast<std::uint16_t>(data[0]) << 8) |
        static_cast<std::uint16_t>(data[1])
    );
}

std::uint32_t read_u32_be(
    const std::uint8_t* data
) {
    return
        (static_cast<std::uint32_t>(data[0]) << 24) |
        (static_cast<std::uint32_t>(data[1]) << 16) |
        (static_cast<std::uint32_t>(data[2]) << 8) |
        static_cast<std::uint32_t>(data[3]);
}

std::uint64_t read_u64_be(
    const std::uint8_t* data
) {
    std::uint64_t value = 0;

    for (std::size_t i = 0; i < 8; ++i) {
        value =
            (value << 8) |
            static_cast<std::uint64_t>(data[i]);
    }

    return value;
}

bool is_valid_state(State state) {
    return state == State::LIVE ||
           state == State::TOMBSTONE;
}
}

std::uint16_t crc16(
    const std::uint8_t* data,
    std::size_t length
) {
    std::uint16_t crc = 0xFFFF;

    for (std::size_t i = 0; i < length; ++i) {
        crc ^= static_cast<std::uint16_t>(data[i]) << 8;

        for (std::uint8_t bit = 0; bit < 8; ++bit) {
            if ((crc & 0x8000u) != 0) {
                crc =
                    static_cast<std::uint16_t>(
                        (crc << 1) ^ 0x1021u
                    );
            } else {
                crc =
                    static_cast<std::uint16_t>(
                        crc << 1
                    );
            }
        }
    }

    return crc;
}
Result serialize(
    const Record& record,
    std::uint8_t* out,
    std::size_t out_capacity,
    std::size_t& bytes_written
) {
    bytes_written = 0;

    if (out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (!is_valid_state(record.state)) {
        return Result::INVALID_ARGUMENT;
    }

    if (record.key.len > KEY_MAX) {
        return Result::KEY_TOO_LARGE;
    }

    if (record.value.len > VALUE_MAX) {
        return Result::VALUE_TOO_LARGE;
    }

    if (record.key.len > 0 && record.key.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (record.value.len > 0 && record.value.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (record.state == State::TOMBSTONE &&
        record.value.len != 0) {
        return Result::INVALID_ARGUMENT;
    }

    const std::size_t total_length =
        HEADER_SIZE +
        record.key.len +
        record.value.len +
        CRC_SIZE +
        COMMIT_MARKER_SIZE;

    if (total_length > MAX_RECORD_SIZE) {
        return Result::INVALID_ARGUMENT;
    }

    if (out_capacity < total_length) {
        return Result::BUFFER_TOO_SMALL;
    }

    const std::uint8_t flags =
        (record.state == State::TOMBSTONE)
            ? FLAGS_TOMBSTONE
            : 0x00;

    write_u16_be(
        out + OFFSET_MAGIC,
        MAGIC
    );

    out[OFFSET_FORMAT_VERSION] = FORMAT_VERSION;
    out[OFFSET_FLAGS] = flags;

    write_u16_be(
        out + OFFSET_RECORD_LEN,
        static_cast<std::uint16_t>(total_length)
    );

    out[OFFSET_KEY_LEN] =
        static_cast<std::uint8_t>(record.key.len);

    write_u16_be(
        out + OFFSET_VALUE_LEN,
        static_cast<std::uint16_t>(record.value.len)
    );

    write_u64_be(
        out + OFFSET_VERSION_HLC,
        record.version.hlc
    );

    out[OFFSET_VERSION_ORIGIN] =
        record.version.origin;

    write_u32_be(
        out + OFFSET_SEQUENCE,
        record.sequence
    );

    if (record.key.len > 0) {
        std::memcpy(
            out + OFFSET_KEY,
            record.key.data,
            record.key.len
        );
    }

    const std::size_t value_offset =
        OFFSET_KEY + record.key.len;

    if (record.value.len > 0) {
        std::memcpy(
            out + value_offset,
            record.value.data,
            record.value.len
        );
    }

    const std::size_t crc_offset =
        value_offset + record.value.len;

    const std::uint16_t checksum =
        crc16(out, crc_offset);

    write_u16_be(
        out + crc_offset,
        checksum
    );

    out[crc_offset + CRC_SIZE] = COMMIT_MARKER;

    bytes_written = total_length;

    return Result::OK;
}

Result deserialize(
    const std::uint8_t* data,
    std::size_t record_len,
    Record& record
) {
    record = {};

    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (record_len < MIN_RECORD_SIZE ||
        record_len > MAX_RECORD_SIZE) {
        return Result::CORRUPTED;
    }

    const std::uint16_t stored_magic =
        read_u16_be(data + OFFSET_MAGIC);

    if (stored_magic != MAGIC) {
        return Result::CORRUPTED;
    }

    if (data[OFFSET_FORMAT_VERSION] != FORMAT_VERSION) {
        return Result::CORRUPTED;
    }

    const std::uint8_t flags =
        data[OFFSET_FLAGS];

    if ((flags & FLAGS_RESERVED_MASK) != 0) {
        return Result::CORRUPTED;
    }

    const std::uint16_t encoded_length =
        read_u16_be(data + OFFSET_RECORD_LEN);

    if (encoded_length != record_len) {
        return Result::CORRUPTED;
    }

    const std::uint8_t key_len =
        data[OFFSET_KEY_LEN];

    if (key_len > KEY_MAX) {
        return Result::CORRUPTED;
    }

    const std::uint16_t value_len =
        read_u16_be(data + OFFSET_VALUE_LEN);

    if (value_len > VALUE_MAX) {
        return Result::CORRUPTED;
    }

    const bool tombstone =
        (flags & FLAGS_TOMBSTONE) != 0;

    if (tombstone && value_len != 0) {
        return Result::CORRUPTED;
    }

    const std::size_t expected_length =
        HEADER_SIZE +
        key_len +
        value_len +
        CRC_SIZE +
        COMMIT_MARKER_SIZE;

    if (expected_length != record_len) {
        return Result::CORRUPTED;
    }

    const std::size_t value_offset =
        OFFSET_KEY + key_len;

    const std::size_t crc_offset =
        value_offset + value_len;

    if (data[record_len - 1] != COMMIT_MARKER) {
        return Result::CORRUPTED;
    }

    const std::uint16_t stored_crc =
        read_u16_be(data + crc_offset);

    const std::uint16_t calculated_crc =
        crc16(data, crc_offset);

    if (stored_crc != calculated_crc) {
        return Result::CORRUPTED;
    }

    record.state =
        tombstone
            ? State::TOMBSTONE
            : State::LIVE;

    record.version.hlc =
        read_u64_be(data + OFFSET_VERSION_HLC);

    record.version.origin =
        data[OFFSET_VERSION_ORIGIN];

    record.sequence =
        read_u32_be(data + OFFSET_SEQUENCE);

    record.key = {
        data + OFFSET_KEY,
        key_len
    };

    record.value = {
        data + value_offset,
        value_len
    };

    return Result::OK;
}

    0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF

QueueHandle_t rx_queue = nullptr;

    RX_QUEUE_DEPTH * sizeof(ReceivedFrame)

static std::uint8_t tx_status_queue_storage[

const std::uint8_t* mac_for_node(node::Id id) {
    switch (id) {
        case node::A:
            return MAC_A;

        case node::B:
            return MAC_B;

        case node::C:
            return MAC_C;

        case node::D:
            return MAC_D;

        default:
            return nullptr;
    }
}

node::Id node_from_mac(const std::uint8_t* mac) {
    if (mac == nullptr) {
        return node::INVALID;
    }

    if (std::memcmp(mac, MAC_A, MAC_ADDRESS_SIZE) == 0) {
        return node::A;
    }

    if (std::memcmp(mac, MAC_B, MAC_ADDRESS_SIZE) == 0) {
        return node::B;
    }

    if (std::memcmp(mac, MAC_C, MAC_ADDRESS_SIZE) == 0) {
        return node::C;
    }

    if (std::memcmp(mac, MAC_D, MAC_ADDRESS_SIZE) == 0) {
        return node::D;
    }

    return node::INVALID;
}

void on_receive(
    const std::uint8_t* mac_addr,
    const std::uint8_t* data,
    int data_len
) {
    if (rx_queue == nullptr ||
        mac_addr == nullptr ||
        data == nullptr ||
        data_len < 0 ||
        data_len > static_cast<int>(MAX_FRAME_SIZE)) {
        return;
    }

    ReceivedFrame frame{};

    frame.length = static_cast<std::uint8_t>(data_len);

    std::memcpy(
        frame.data,
        data,
        static_cast<std::size_t>(data_len)
    );

    std::memcpy(
        frame.source_mac,
        mac_addr,
        MAC_ADDRESS_SIZE
    );

    // Never block the ESP-NOW callback.
    (void)xQueueSend(
        rx_queue,
        &frame,
        0
    );
}
void on_send(
    const std::uint8_t* mac_addr,
    esp_now_send_status_t status
) {
    if (tx_status_queue == nullptr ||
        mac_addr == nullptr) {
        return;
    }

    const node::Id destination = node_from_mac(mac_addr);

    if (destination == node::INVALID) {
        return;

    SendStatus event{

        status == ESP_NOW_SEND_SUCCESS
    };

    // Never block the callback.
    (void)xQueueSend(
        tx_status_queue,
        &event,
        0
    );
}

Result EspNowTransport::begin(
    node::Id local_node
) {
    if (local_node < node::MIN ||
        local_node > node::MAX) {
        return Result::INVALID_NODE;
    }

    local_node_id = local_node;

    rx_queue = xQueueCreateStatic(
        RX_QUEUE_DEPTH,
        sizeof(ReceivedFrame),
        rx_queue_storage,
        &rx_queue_control
    );

    tx_status_queue = xQueueCreateStatic(
        RX_QUEUE_DEPTH,
        sizeof(SendStatus),
        tx_status_queue_storage,
        &tx_status_queue_control
    );

    if (rx_queue == nullptr ||
        tx_status_queue == nullptr) {
        return Result::SEND_FAILED;
    }

    WiFi.mode(WIFI_STA);
    WiFi.setSleep(false);
    WiFi.setTxPower(WIFI_POWER_8_5dBm);

    const esp_err_t channel_result =
        esp_wifi_set_channel(
            CHANNEL,
            WIFI_SECOND_CHAN_NONE
        );

    if (channel_result != ESP_OK) {
        return Result::SEND_FAILED;
    }

    if (esp_now_init() != ESP_OK) {
        return Result::SEND_FAILED;
    }

    esp_now_register_recv_cb(on_receive);
    esp_now_register_send_cb(on_send);

        for (node::Id id = node::MIN;
         id <= node::MAX;
         ++id) {

        if (id == local_node_id) {
            continue;
        }

        const std::uint8_t* mac = mac_for_node(id);

        if (mac == nullptr) {
            return Result::INVALID_NODE;
        }

        esp_now_peer_info_t peer_info = {};

        std::memcpy(
            peer_info.peer_addr,
            mac,
            MAC_ADDRESS_SIZE
        );

        peer_info.channel = CHANNEL;
        peer_info.encrypt = false;
        peer_info.ifidx = WIFI_IF_STA;

        const esp_err_t result =
            esp_now_add_peer(&peer_info);

        if (result != ESP_OK) {
            return Result::SEND_FAILED;
        }
    }

    /*
     * Register the broadcast address as an ESP-NOW peer.
     * Broadcast transmission uses this MAC directly.
     */
    esp_now_peer_info_t broadcast_peer = {};

    std::memcpy(
        broadcast_peer.peer_addr,
        MAC_BROADCAST,
        MAC_ADDRESS_SIZE
    );

    broadcast_peer.channel = CHANNEL;
    broadcast_peer.encrypt = false;
    broadcast_peer.ifidx = WIFI_IF_STA;

    const esp_err_t broadcast_result =
        esp_now_add_peer(
            &broadcast_peer
        );

    if (broadcast_result != ESP_OK) {
        return Result::SEND_FAILED;
    }

    initialized = true;

    return Result::OK;
}

Result EspNowTransport::send_unicast(
    node::Id destination,
    const std::uint8_t* data,
    std::size_t length
) {
    if (!initialized) {
        return Result::NOT_INITIALIZED;
    }

    if (destination < node::MIN ||
        destination > node::MAX) {
        return Result::INVALID_NODE;
    }

    if (destination == local_node_id) {
        return Result::INVALID_NODE;
    }

    if (length > MAX_FRAME_SIZE) {
        return Result::FRAME_TOO_LARGE;
    }

    if (length > 0 && data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    const std::uint8_t* mac = mac_for_node(destination);

    if (mac == nullptr) {
        return Result::PEER_NOT_FOUND;
    }

    const esp_err_t result =
        esp_now_send(
            mac,
            data,
            length
        );

    if (result != ESP_OK) {
        if (result == ESP_ERR_ESPNOW_NOT_FOUND) {
            return Result::PEER_NOT_FOUND;
        }

        return Result::SEND_FAILED;
    }

    return Result::OK;
}


Result EspNowTransport::send_broadcast(
    const std::uint8_t* data,
    std::size_t length
) {
    if (!initialized) {
        return Result::NOT_INITIALIZED;
    }

    if (length > MAX_FRAME_SIZE) {
        return Result::FRAME_TOO_LARGE;
    }

    if (length > 0 && data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    const esp_err_t result =
        esp_now_send(
            MAC_BROADCAST,
            data,
            length
        );

    if (result != ESP_OK) {
        return Result::SEND_FAILED;
    }

    return Result::OK;
}

#include <cassert>
#include <cstdint>
#include <iostream>

#include "hlc.h"

using namespace hlc;

static std::uint64_t fake_now_ms = 0;
static bool advance_before_next_return = false;
static std::uint64_t advance_to_ms = 0;

static std::uint64_t fake_physical_now()
{
    if (advance_before_next_return) {
        fake_now_ms = advance_to_ms;
        advance_before_next_return = false;
    }

    return fake_now_ms;
}

static void test_encoding()
{
    const Hlc value =
        make(
            0x123456789ABCuLL,
            0xBEEF
        );

    assert(
        physical(value) ==
        0x123456789ABCuLL
    );

    assert(
        logical(value) ==
        0xBEEF
    );
}

static void test_comparison()
{
    assert(
        compare(
            make(100, 1),
            make(100, 2)
        ) < 0
    );

    assert(
        compare(
            make(101, 0),
            make(100, 0)
        ) > 0
    );

    assert(
        compare(
            make(500, 9),
            make(500, 9)
        ) == 0
    );
}

static void test_reboot_anchor()
{
    fake_now_ms = 100;

    Clock clock(fake_physical_now);

    assert(
        clock.begin(
            make(1000, 7)
        )
    );

    assert(clock.ready());

    Hlc value = clock.now();

    /*
     * Raw timer = 100 ms.
     * Persisted physical = 1000 ms.
     * Runtime offset must reconstruct physical = 1000.
     */
    assert(physical(value) == 1000);
    assert(logical(value) == 7);

    /*
     * Same physical millisecond:
     * logical counter increments.
     */
    value = clock.on_local_event();

    assert(physical(value) == 1000);
    assert(logical(value) == 8);

    /*
     * Raw clock advances by 1 ms.
     * Reconstructed physical clock advances too.
     */
    fake_now_ms = 101;

    value = clock.on_local_event();

    assert(physical(value) == 1001);
    assert(logical(value) == 0);
}

static void test_receive_rules()
{
    fake_now_ms = 1000;

    Clock clock(fake_physical_now);

    assert(
        clock.begin(
            make(1000, 4)
        )
    );

    /*
     * p == local == remote
     */
    Hlc value =
        clock.on_receive(
            make(1000, 9)
        );

    assert(physical(value) == 1000);
    assert(logical(value) == 10);

    /*
     * physical_now is larger than both.
     */
    fake_now_ms = 1001;

    value =
        clock.on_receive(
            make(900, 50)
        );

    assert(physical(value) == 1001);
    assert(logical(value) == 0);

    /*
     * p == remote
     */
    value =
        clock.on_receive(
            make(1100, 50)
        );

    assert(physical(value) == 1100);
    assert(logical(value) == 51);
}

{
    fake_now_ms = 100;

    Clock clock(fake_physical_now);

    assert(
        clock.begin(
            make(500, LOGICAL_MAX)
        )
    );

    /*
     * First call observes the same physical millisecond,
     * so logical counter is already at maximum.
     *
     * The next physical-time read advances to raw 101,
     * giving reconstructed physical 501.
     */
    advance_before_next_return = true;
    advance_to_ms = 101;

    Hlc value =
        clock.on_local_event();

    assert(physical(value) == 501);
    assert(logical(value) == 0);
}

#include <cassert>
#include <cstddef>
#include <cstdint>
#include <cstdio>

#include "membership.h"
#include "placement.h"

static membership::Manager make_membership(
    node::Id local,
    std::uint8_t live_mask,
    std::uint64_t now_ms
)
{
    membership::Manager manager(
        local,
        1000
    );

    assert(manager.begin(now_ms));

    for (
        node::Id id = node::MIN;
        id <= node::MAX;
        ++id
    ) {
        const std::uint8_t bit =
            static_cast<std::uint8_t>(1u << (id - 1));

        if (
            id != local &&
            (live_mask & bit) != 0
        ) {
            assert(
                manager.heartbeat_received(
                    id,
                    now_ms
                )
            );
        }
    }

    return manager;
}

static std::size_t count_live(
    std::uint8_t live_mask
)
{
    std::size_t count = 0;

    for (
        node::Id id = node::MIN;
        id <= node::MAX;
        ++id
    ) {
        if (
            live_mask &
            static_cast<std::uint8_t>(
                1u << (id - 1)
            )
        ) {
            ++count;
        }
    }

    return count;
}

static bool is_live(
    node::Id id,
    std::uint8_t live_mask
)
{
    return (
        live_mask &
        static_cast<std::uint8_t>(
            1u << (id - 1)
        )
    ) != 0;
}

int main()
{
    placement::Resolver resolver;

    assert(resolver.begin());
    assert(resolver.ready());

    constexpr std::uint64_t NOW = 1000;

    /*
     * Test every possible non-empty membership view.
     *
     * 1  = A alive
     * 2  = B alive
     * 4  = C alive
     * 8  = D alive
     *
     * Therefore masks 1..15 cover every possible
     * non-empty live-node combination.
     */
    for (
        std::uint8_t live_mask = 1;
        live_mask <= 15;
        ++live_mask
    ) {
        const std::size_t live_count =
            count_live(live_mask);

        for (
            std::uint32_t key_index = 0;
            key_index < 1000;
            ++key_index
        ) {
            const std::uint8_t key[] = {
                'k',
                static_cast<std::uint8_t>(
                    key_index & 0xFF
                ),
                static_cast<std::uint8_t>(
                    (key_index >> 8) & 0xFF
                ),
                static_cast<std::uint8_t>(
                    live_mask
                )
            };

            bool expected_set = false;
            placement::Assignment expected{};

            /*
             * Every alive node sees the same membership
             * view and must calculate the same placement
             * for this exact key.
             */
            for (
                node::Id local = node::MIN;
                local <= node::MAX;
                ++local
            ) {
                if (!is_live(local, live_mask)) {
                    continue;
                }

                membership::Manager membership =
                    make_membership(
                        local,
                        live_mask,
                        NOW
                    );

                placement::Assignment assignment{};

                assert(
                    resolver.locate(
                        key,
                        sizeof(key),
                        membership,
                        NOW,
                        assignment
                    )
                );

                assert(
                    is_live(
                        assignment.primary,
                        live_mask
                    )
                );

                if (live_count == 1) {
                    assert(
                        assignment.replica ==
                        node::INVALID
                    );
                } else {
                    assert(
                        assignment.replica !=
                        node::INVALID
                    );

                    assert(
                        assignment.replica !=
                        assignment.primary
                    );

                    assert(
                        is_live(
                            assignment.replica,
                            live_mask
                        )
                    );
                }

                if (!expected_set) {
                    expected = assignment;
                    expected_set = true;
                } else {
                    assert(
                        assignment.primary ==
                        expected.primary
                    );

                    assert(
                        assignment.replica ==
                        expected.replica
                    );
                }
            }
        }
    }

    /*
     * Failure transition tests.
     *
     * Start with all four nodes alive.
     */
    {
        membership::Manager all_alive =
            make_membership(
                node::A,
                0x0F,
                NOW
            );

        const std::uint8_t key[] = {
            'f', 'a', 'i', 'l'
        };

        placement::Assignment before{};

        assert(
            resolver.locate(
                key,
                sizeof(key),
                all_alive,
                NOW,
                before
            )
        );

        assert(before.primary != node::INVALID);
        assert(before.replica != node::INVALID);
        assert(before.primary != before.replica);

        /*
         * Kill the primary.
         *
         * With RF=2, the old replica is the first
         * different live node clockwise, so it must
         * be promoted to primary.
         */
        {
            const std::uint8_t live_mask =
                static_cast<std::uint8_t>(
                    0x0F &
                    ~(
                        1u <<
                        (before.primary - 1)
                    )
                );

            assert(count_live(live_mask) == 3);

            const node::Id local =
                before.primary == node::A
                    ? node::B
                    : node::A;

            assert(is_live(local, live_mask));

            membership::Manager membership =
                make_membership(
                    local,
                    live_mask,
                    NOW
                );

            placement::Assignment after{};

            assert(
                resolver.locate(
                    key,
                    sizeof(key),
                    membership,
                    NOW,
                    after
                )
            );

            assert(
                after.primary ==
                before.replica
            );

            assert(
                after.primary !=
                before.primary
            );

            assert(
                after.replica !=
                node::INVALID
            );

            assert(
                after.replica !=
                after.primary
            );

            assert(
                is_live(
                    after.replica,
                    live_mask
                )
            );
        }

        /*
         * Kill the replica.
         *
         * The primary must remain unchanged.
         * A different live node becomes replica.
         */
        {
            const std::uint8_t live_mask =
                static_cast<std::uint8_t>(
                    0x0F &
                    ~(
                        1u <<
                        (before.replica - 1)
                    )
                );

            assert(count_live(live_mask) == 3);

            node::Id local = node::INVALID;

            for (
                node::Id candidate = node::MIN;
                candidate <= node::MAX;
                ++candidate
            ) {
                if (is_live(candidate, live_mask)) {
                    local = candidate;
                    break;
                }
            }

            assert(local != node::INVALID);

            membership::Manager membership =
                make_membership(
                    local,
                    live_mask,
                    NOW
                );

            placement::Assignment after{};

            assert(
                resolver.locate(
                    key,
                    sizeof(key),
                    membership,
                    NOW,
                    after
                )
            );

            assert(
                after.primary ==
                before.primary
            );

            assert(
                after.replica !=
                node::INVALID
            );

            assert(
                after.replica !=
                before.replica
            );

            assert(
                after.replica !=
                after.primary
            );

            assert(
                is_live(
                    after.replica,
                    live_mask
                )
            );
        }
    }

    /*
     * One-node case.
     *
     * Placement still succeeds, but RF=2 cannot
     * be satisfied, so replica is INVALID.
     */
    {
        membership::Manager only_a(
            node::A,
            1000
        );

        assert(only_a.begin(NOW));

        const std::uint8_t key[] = {
            'o', 'n', 'e'
        };

        placement::Assignment result{};

        assert(
            resolver.locate(
                key,
                sizeof(key),
                only_a,
                NOW,
                result
            )
        );

        assert(result.primary == node::A);
        assert(result.replica == node::INVALID);
    }


    /*
     * Distribution sanity test.
     *
     * With all nodes alive, sample many keys and count
     * primary assignments. This does not require exact
     * percentages; it detects pathological skew.
     */
    {
        membership::Manager membership =
            make_membership(
                node::A,
                0x0F,
                NOW
            );

        std::size_t primary_counts[5]{};

        constexpr std::uint32_t KEY_COUNT = 100000;

        for (
            std::uint32_t i = 0;
            i < KEY_COUNT;
            ++i
        ) {
            const std::uint8_t key[] = {
                'd',
                static_cast<std::uint8_t>(
                    i & 0xFF
                ),
                static_cast<std::uint8_t>(
                    (i >> 8) & 0xFF
                ),
                static_cast<std::uint8_t>(
                    (i >> 16) & 0xFF
                ),
                static_cast<std::uint8_t>(
                    (i >> 24) & 0xFF
                )
            };

            placement::Assignment assignment{};

            assert(
                resolver.locate(
                    key,
                    sizeof(key),
                    membership,
                    NOW,
                    assignment
                )
            );

            ++primary_counts[assignment.primary];
        }

        std::size_t minimum = KEY_COUNT;
        std::size_t maximum = 0;

        for (
            node::Id id = node::MIN;
            id <= node::MAX;
            ++id
        ) {
            assert(primary_counts[id] > 0);

            if (primary_counts[id] < minimum) {
                minimum = primary_counts[id];
            }

            if (primary_counts[id] > maximum) {
                maximum = primary_counts[id];
            }
        }

        assert(maximum <= minimum * 3);

        std::printf(
            "Distribution: A=%zu B=%zu C=%zu D=%zu\n",
            primary_counts[node::A],
            primary_counts[node::B],
            primary_counts[node::C],
            primary_counts[node::D]
        );
    }

    return 0;
}

        0x02,
        0x16,

    };

    for (std::size_t i = 0; i < HEADER_SIZE; ++i) {

    }

    PacketHeader decoded{};

    result = deserialize_header(
        buffer,
        HEADER_SIZE + original.payload_len,
        decoded
    );

    assert(result == Result::OK);

    assert(decoded.protocol_version == original.protocol_version);
    assert(decoded.msg_type == original.msg_type);
    assert(decoded.flags == original.flags);
    assert(decoded.header_len == original.header_len);
    assert(decoded.request_id == original.request_id);
    assert(decoded.origin_node_id == original.origin_node_id);
    assert(decoded.sender_node_id == original.sender_node_id);
    assert(decoded.destination_node_id == original.destination_node_id);
    assert(decoded.ttl == original.ttl);
    assert(decoded.hlc == original.hlc);
    assert(decoded.payload_len == original.payload_len);

        auto expect_invalid = [](
        std::uint8_t* buffer,
        std::size_t packet_len
    ) {
        PacketHeader decoded{};

        Result result = deserialize_header(
            buffer,
            packet_len,
            decoded
        );

        assert(result == Result::INVALID_PACKET);
    };

    // Wrong protocol version.
    buffer[0] = 0x02;
    expect_invalid(buffer, HEADER_SIZE + original.payload_len);
    buffer[0] = PROTOCOL_VERSION;

    // Invalid header length.
    buffer[3] = 21;
    expect_invalid(buffer, HEADER_SIZE + original.payload_len);
    buffer[3] = HEADER_SIZE;

    // Invalid origin node ID.
    buffer[8] = NODE_ID_INVALID;
    expect_invalid(buffer, HEADER_SIZE + original.payload_len);
    buffer[8] = NODE_A;

    // Invalid sender node ID.
    buffer[9] = 0x05;
    expect_invalid(buffer, HEADER_SIZE + original.payload_len);
    buffer[9] = NODE_A;

    // Invalid destination node ID.
    buffer[10] = 0x05;
    expect_invalid(buffer, HEADER_SIZE + original.payload_len);
    buffer[10] = NODE_B;

    // Non-zero TTL is forbidden in v1.
    buffer[11] = 0x01;
    expect_invalid(buffer, HEADER_SIZE + original.payload_len);
    buffer[11] = 0x00;

    // Reserved flag bit is set.
    buffer[2] = 0x08;
    expect_invalid(buffer, HEADER_SIZE + original.payload_len);
    buffer[2] = FLAG_RETRY;

    // Unknown message type.
    buffer[1] = 0x7F;
    expect_invalid(buffer, HEADER_SIZE + original.payload_len);
    buffer[1] = static_cast<std::uint8_t>(MessageType::SET_REQ);

    // Declared payload is larger than the v1 maximum.
    buffer[20] = 0x00;
    buffer[21] = 0xE5; // 229
    expect_invalid(buffer, MAX_PACKET_SIZE);

    // Declared payload length does not match the actual packet length.
    buffer[20] = 0x00;
    buffer[21] = 0x05;
    expect_invalid(buffer, HEADER_SIZE + 4);

    // Serialization buffer too small.
    std::uint8_t small_buffer[HEADER_SIZE - 1]{};

    result = serialize_header(
        original,
        small_buffer,
        sizeof(small_buffer)
    );

    assert(result == Result::BUFFER_TOO_SMALL);

    // Null output buffer.
    result = serialize_header(
        original,
        nullptr,
        HEADER_SIZE
    );

    assert(result == Result::INVALID_ARGUMENT);

    std::cout << "Header validation tests: PASS\n";
    std::cout << "Header round-trip test: PASS\n";

    return 0;
}

    Result result = serialize_packet(
        header,
        payload,
        buffer,
        sizeof(buffer),
        bytes_written
    );

    assert(result == Result::OK);
    assert(bytes_written == HEADER_SIZE + 8);

    // Header bytes.
    assert(buffer[0] == PROTOCOL_VERSION);
    assert(buffer[1] ==
           static_cast<std::uint8_t>(MessageType::SET_REQ));

    // Big-endian request ID: 42 = 00 00 00 2A.
    assert(buffer[4] == 0x00);
    assert(buffer[5] == 0x00);
    assert(buffer[6] == 0x00);
    assert(buffer[7] == 0x2A);

    // Payload starts exactly after the 22-byte header.
    assert(std::memcmp(
        buffer + HEADER_SIZE,
        payload_data,
        sizeof(payload_data)
    ) == 0);

    // ------------------------------------------------------------
    // Packet round-trip
    // ------------------------------------------------------------

    PacketView decoded{};

    result = deserialize_packet(
        buffer,
        bytes_written,
        decoded
    );

    assert(result == Result::OK);

    assert(decoded.header.request_id == 42);
    assert(decoded.header.origin_node_id == NODE_A);
    assert(decoded.header.sender_node_id == NODE_A);
    assert(decoded.header.destination_node_id == NODE_B);
    assert(decoded.header.hlc == 0x0001020304050607ULL);
    assert(decoded.header.payload_len == 8);

    assert(decoded.payload.len == 8);
    assert(std::memcmp(
        decoded.payload.data,
        payload_data,
        8
    ) == 0);

    std::cout << "Packet round-trip test: PASS\n";

    // ------------------------------------------------------------
    // Header/payload length mismatch
    // ------------------------------------------------------------

    PacketHeader bad_header = header;
    bad_header.payload_len = 7;

    result = serialize_packet(
        bad_header,
        payload,
        buffer,
        sizeof(buffer),
        bytes_written
    );

    assert(result == Result::INVALID_PACKET);

    std::cout << "Packet length consistency test: PASS\n";

    // ------------------------------------------------------------

    );

    manager.tick(3499);

        .origin = node::B
    };

    assert(compare_versions(origin_a, origin_b) < 0);

    // Test HLC equality with the largest valid node ID.
    Version origin_c{
        .hlc = 3000,
        .origin = node::C
    };

    Version origin_d{
        .hlc = 3000,
        .origin = node::D
    };

    assert(compare_versions(origin_c, origin_d) < 0);

    // Sanity: comparison must be antisymmetric for these cases.
    assert(
        compare_versions(older_hlc, newer_hlc) ==
        -compare_versions(newer_hlc, older_hlc)
    );

    assert(
        compare_versions(lower_origin, higher_origin) ==
        -compare_versions(higher_origin, lower_origin)
    );

    std::cout << "Storage version comparison tests: PASS\n";

    return 0;
}
