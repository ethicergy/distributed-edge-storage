/*
 * This intentionally drops the first SET_ACK generated by
 * the receiver.
 *
 * Expected sequence:
 *
 *     B -> A : SET_REQ
 *     A       : execute exactly once
 *     A       : cache SET_ACK
 *     A       : drop first SET_ACK
 *     B       : timeout
 *     B -> A : RETRY SET_REQ
 *     A       : DEDUP DUPLICATE_COMPLETED
 *     A -> B : cached SET_ACK
 *     B       : request completes
 */
bool drop_first_set_ack = true;

std::uint64_t client_start_time_ms = 0;

bool build_set_request(
    request::Identity identity,
    const std::uint8_t* key,
    std::size_t key_length,
    const std::uint8_t* value,
    std::size_t value_length,
    std::uint8_t* output,
    std::size_t output_capacity,
    std::size_t& output_length
)
{
    if (
        key == nullptr ||
        value == nullptr ||
        output == nullptr
    ) {
        return false;
    }

    if (key_length == 0 ||
        key_length > storage::KEY_MAX) {
        return false;
    }

    if (value_length > storage::VALUE_MAX) {
        return false;
    }

    protocol::SetRequestPayload payload{
        protocol::KeyView{
            key,
            static_cast<std::uint8_t>(key_length)
        },
        protocol::ValueView{
            value,
            static_cast<std::uint8_t>(value_length)
        }
    };

    std::uint8_t payload_buffer[
        protocol::MAX_PAYLOAD_SIZE
    ]{};

    std::size_t payload_length = 0;

    if (
        protocol::serialize_set_request(
            payload,
            payload_buffer,
            sizeof(payload_buffer),
            payload_length
        ) != protocol::Result::OK
    ) {
        return false;
    }

    const std::uint64_t mutation_hlc =
        hlc_clock.on_local_event();

    const protocol::PacketHeader header{
        protocol::PROTOCOL_VERSION,
        protocol::MessageType::SET_REQ,
        0,
        protocol::HEADER_SIZE,

        identity.request_id,

        identity.origin,
        LOCAL_NODE_ID,
        TEST_PEER_NODE_ID,

        0,

        mutation_hlc,

        static_cast<std::uint16_t>(
            payload_length
        )
    };

    return
        protocol::serialize_packet(
            header,
            protocol::ByteView{
                payload_buffer,
                payload_length
            },
            output,
            output_capacity,
            output_length
        ) == protocol::Result::OK;
}

bool reliable_send(
    void* context,
    node::Id destination,
    const std::uint8_t* packet,
    std::size_t packet_length,
    bool retry
)
{
    auto* sender =
        static_cast<RequestSenderContext*>(
            context
        );

    if (
        sender == nullptr ||
        sender->transport == nullptr ||
        packet == nullptr
    ) {
        return false;
    }

    if (
        destination < node::MIN ||
        destination > node::MAX
    ) {
        return false;
    }

    if (!retry) {

        return
            sender->transport->send_unicast(
                destination,
                packet,
                packet_length
            ) == transport::Result::OK;
    }

    std::uint8_t retry_packet[
        protocol::MAX_PACKET_SIZE
    ]{};

    std::size_t retry_length = 0;

    if (
        request::prepare_retry(
            packet,
            packet_length,
            retry_packet,
            sizeof(retry_packet),
            retry_length
        ) != request::PacketResult::OK
    ) {
        return false;
    }

    Serial.print("RETRY SEND -> NODE ");
    Serial.println(
        static_cast<unsigned>(destination)
    );

    return
        sender->transport->send_unicast(
            destination,
            retry_packet,
            retry_length
        ) == transport::Result::OK;
}

bool find_key_owned_by_node(
    node::Id desired_primary
)
{
    /*
     * The test uses only A and C as live nodes.
     * This gives both boards the same routing view.
     */

    const std::uint32_t boot_tag = esp_random();

    for (unsigned index = 0; index < 1000; ++index) {

        const int written = std::snprintf(
            reinterpret_cast<char*>(test_key),
            sizeof(test_key),
            "phase6_wire_%lu_%03u",
            static_cast<unsigned long>(boot_tag),
            index
        );

        if (written <= 0) {
            continue;
        }

        const std::size_t length =
            static_cast<std::size_t>(written);

        if (length > storage::KEY_MAX) {
            continue;
        }

        placement::Assignment assignment{};

        if (
            !placement_resolver.locate(
                test_key,
                length,
                membership_manager,
                static_cast<std::uint64_t>(millis()),
                assignment
            )
        ) {
            continue;
        }

        if (assignment.primary != desired_primary) {
            continue;
        }

        test_key_length = length;

        Serial.print("TEST KEY: ");

        for (std::size_t i = 0; i < test_key_length; ++i) {
            Serial.write(test_key[i]);
        }

        Serial.println();

        Serial.print("PRIMARY: ");
        Serial.println(
            static_cast<unsigned>(assignment.primary)
        );

        Serial.print("REPLICA: ");
        Serial.println(
            static_cast<unsigned>(assignment.replica)
        );

        return true;
    }

    return false;
}

void process_server_request(
    const transport::ReceivedFrame& frame
)
{
    protocol::PacketView packet{};

    if (
        protocol::deserialize_packet(
            frame.data,
            frame.length,
            packet
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "RX PACKET DECODE: FAIL"
        );
        return;
    }

    if (
        packet.header.destination_node_id !=
        LOCAL_NODE_ID
    ) {
        return;
    }

    if (
        packet.header.msg_type !=
            protocol::MessageType::SET_REQ &&
        packet.header.msg_type !=
            protocol::MessageType::GET_REQ &&
        packet.header.msg_type !=
            protocol::MessageType::DEL_REQ
    ) {
        return;
    }

    protocol::DecodedMessage decoded{};

    if (
        protocol::decode_message(
            packet,
            decoded
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "APPLICATION DECODE: FAIL"
        );
        return;
    }

    request::Identity identity{};

    if (
        request::extract_identity(
            packet,
            identity
        ) != request::PacketResult::OK
    ) {
        Serial.println(
            "REQUEST IDENTITY: FAIL"
        );
        return;
    }

    /*
     * Merge remote HLC before application processing.
     */
    const hlc::Hlc merged_hlc =
        hlc_clock.on_receive(
            packet.header.hlc
        );

    if (merged_hlc == 0) {
        Serial.println(
            "RX HLC MERGE: REJECTED"
        );
        return;
    }

    const std::uint64_t now =
        static_cast<std::uint64_t>(millis());

    const request::DedupResult dedup_result =
        dedup_manager.begin_request(
            identity,
            now
        );

    if (
        dedup_result ==
        request::DedupResult::NEW_REQUEST
    ) {

        Serial.println();
        Serial.println(
            "SERVER: NEW REQUEST"
        );

        kv::OperationResponse operation_response{};

        const kv::Result operation_result =
            kv_engine.execute(
                decoded,
                membership_manager,
                now,
                operation_response
            );

        Serial.print(
            "KV ENGINE RESULT: "
        );

        Serial.println(
            static_cast<unsigned>(
                operation_result
            )
        );

        if (
            packet.header.msg_type ==
            protocol::MessageType::SET_REQ
        ) {
            ++receiver_set_execution_count;

            Serial.print(
                "SET EXECUTION COUNT: "
            );

            Serial.println(
                receiver_set_execution_count
            );
        }

        std::uint8_t response[
            protocol::MAX_PACKET_SIZE
        ]{};

        std::size_t response_length = 0;

        const hlc::Hlc response_hlc =
            hlc_clock.on_local_event();

        if (
            !kv::ResponseBuilder::build(
                packet.header,
                LOCAL_NODE_ID,
                response_hlc,
                operation_response,
                response,
                sizeof(response),
                response_length
            )
        ) {
                        Serial.println(
                "RESPONSE BUILD: FAIL"
            );
            dedup_manager.remove(identity);
            return;
        }

        const request::DedupResult cache_result =
            dedup_manager.complete_request(
                identity,
                response,
                response_length,
                now
            );

        if (
            cache_result !=
            request::DedupResult::NEW_REQUEST
        ) {
            Serial.print(
                "DEDUP CACHE: FAIL ("
            );

            Serial.print(
                static_cast<unsigned>(
                    cache_result
                )
            );

            Serial.println(")");

            return;
        }

        Serial.println(
            "DEDUP CACHE: COMPLETED"
        );

        /*
         * Intentionally lose the first SET_ACK.
         */
        if (
            packet.header.msg_type ==
                protocol::MessageType::SET_REQ &&
            drop_first_set_ack
        ) {
            drop_first_set_ack = false;

            Serial.println(
                "SET_ACK: DELIBERATELY DROPPED"
            );

            return;
        }

        const transport::Result send_result =
            espnow_transport.send_unicast(
                packet.header.origin_node_id,
                response,
                response_length
            );

        if (
            send_result ==
            transport::Result::OK
        ) {
            Serial.println(
                "RESPONSE: SENT"
            );
        } else {
            Serial.print(
                "RESPONSE SEND: FAIL ("
            );

            Serial.print(
                static_cast<unsigned>(
                    send_result
                )
            );

            Serial.println(")");
        }

        return;
    }

    if (
        dedup_result ==
        request::DedupResult::DUPLICATE_IN_PROGRESS
    ) {

        Serial.println(
            "DEDUP: DUPLICATE IN_PROGRESS - SUPPRESSED"
        );

        return;
    }

    if (
        dedup_result ==
        request::DedupResult::DUPLICATE_COMPLETED
    ) {

        Serial.println(
            "DEDUP: DUPLICATE_COMPLETED - REPLAY"
        );

        std::uint8_t cached_response[
            protocol::MAX_PACKET_SIZE
        ]{};

        std::size_t cached_length = 0;

        const request::DedupResult cache_result =
            dedup_manager.get_cached_response(
                identity,
                cached_response,
                sizeof(cached_response),
                cached_length,
                now
            );

        if (
            cache_result !=
            request::DedupResult::DUPLICATE_COMPLETED
        ) {
            Serial.println(
                "DEDUP CACHE READ: FAIL"
            );
            return;
        }

        const transport::Result send_result =
            espnow_transport.send_unicast(
                packet.header.origin_node_id,
                cached_response,
                cached_length
            );

                if (
            send_result ==
            transport::Result::OK
        ) {
            Serial.println(
                "CACHED RESPONSE: REPLAYED"
            );

            Serial.println(
                receiver_set_execution_count == 1
                    ? "EXACTLY-ONCE: PASS"
                    : "EXACTLY-ONCE: FAIL"
            );

            // Re-arm so the next run exercises the retry path again.
            receiver_set_execution_count = 0;
            drop_first_set_ack = true;
        } else {
            Serial.print(
                "CACHED RESPONSE SEND: FAIL ("
            );

            Serial.print(
                static_cast<unsigned>(
                    send_result
                )
            );

            Serial.println(")");
        }

        return;
    }

    Serial.print(
        "DEDUP: UNEXPECTED RESULT ("
    );

    Serial.print(
        static_cast<unsigned>(
            dedup_result
        )
    );

    Serial.println(")");
}

void process_requester_response(
    const transport::ReceivedFrame& frame
)
{
    protocol::PacketView packet{};

    if (
        protocol::deserialize_packet(
            frame.data,
            frame.length,
            packet
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "RESPONSE PACKET DECODE: FAIL"
        );
        return;
    }

    if (
        packet.header.destination_node_id !=
        LOCAL_NODE_ID
    ) {
        return;
    }

    if (
        packet.header.msg_type !=
        protocol::MessageType::SET_ACK
    ) {
        return;
    }

    protocol::DecodedMessage decoded{};

    if (
        protocol::decode_message(
            packet,
            decoded
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "SET_ACK DECODE: FAIL"
        );
        return;
    }

    const hlc::Hlc merged_hlc =
        hlc_clock.on_receive(
            packet.header.hlc
        );

    if (merged_hlc == 0) {
        Serial.println(
            "ACK HLC MERGE: REJECTED"
        );
        return;
    }

    request::Identity identity{
        packet.header.origin_node_id,
        packet.header.request_id
    };

    if (!identity.valid()) {
        Serial.println(
            "ACK IDENTITY: FAIL"
        );
        return;
    }

    const request::Result completion_result =
        request_manager.complete(
            identity,
            frame.data,
            frame.length
        );

    if (
        completion_result ==
        request::Result::OK
    ) {
        request_finished = true;

        Serial.println();
        Serial.println(
            "REQUEST COMPLETION: PASS"
        );

        return;
    }

    if (
        completion_result ==
            request::Result::UNKNOWN_REQUEST ||
        completion_result ==
            request::Result::ALREADY_TERMINAL
    ) {
        Serial.println(
            "RESPONSE: LATE/UNKNOWN - IGNORED"
        );
        return;
    }

    Serial.print(
        "REQUEST COMPLETION: FAIL ("
    );

    Serial.print(
        static_cast<unsigned>(
            completion_result
        )
    );

    Serial.println(")");
}

void print_final_result()
{
    if (
        final_result_printed ||
        !request_finished
    ) {
        return;
    }

    final_result_printed = true;

    request::State state{};

    if (
        request_manager.get_state(
            active_request,
            state
        ) != request::Result::OK
    ) {
        Serial.println(
            "FINAL STATE READ: FAIL"
        );
        return;
    }

    request::RequestInfo info{};

    if (
        request_manager.get_info(
            active_request,
            info
        ) != request::Result::OK
    ) {
        Serial.println(
            "FINAL INFO READ: FAIL"
        );
        return;
    }

    std::uint8_t response[
        protocol::MAX_PACKET_SIZE
    ]{};

    std::size_t response_length = 0;

    if (
        request_manager.get_completion(
            active_request,
            response,
            sizeof(response),
            response_length
        ) != request::Result::OK
    ) {
        Serial.println(
            "FINAL RESPONSE READ: FAIL"
        );
        return;
    }

    protocol::PacketView packet{};

    if (
        protocol::deserialize_packet(
            response,
            response_length,
            packet
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "FINAL RESPONSE PACKET: FAIL"
        );
        return;
    }

    protocol::DecodedMessage decoded{};

    if (
        protocol::decode_message(
            packet,
            decoded
        ) != protocol::Result::OK
    ) {
        Serial.println(
            "FINAL RESPONSE DECODE: FAIL"
        );
        return;
    }

    const bool status_ok =
        decoded.payload.set_ack.status ==
        protocol::StatusCode::OK;

    const bool retry_happened =
    info.retry_count > 0;

    Serial.println();
    Serial.println(
        "========================================"
    );
    Serial.println(
        "PHASE 6 DISTRIBUTED SET RESULT"
    );
    Serial.println(
        "========================================"
    );

    Serial.print(
        "FINAL REQUEST STATE: "
    );

    Serial.println(
        state == request::State::COMPLETED
            ? "COMPLETED"
            : "NOT COMPLETED"
    );

    Serial.print(
        "SET_ACK STATUS: "
    );

    Serial.println(
        status_ok
            ? "OK"
            : "NOT OK"
    );

    Serial.print(
        "RETRY COUNT: "
    );

    Serial.println(
        static_cast<unsigned>(
            info.retry_count
        )
    );

    Serial.print(
        "RETRY PATH: "
    );

    Serial.println(
        retry_happened
            ? "EXERCISED"
            : "NOT EXERCISED"
    );

        Serial.println(
        "EXACTLY-ONCE: check node A log"
    );

    if (
        state == request::State::COMPLETED &&
        status_ok &&
        retry_happened
    ) {
        Serial.println();
        Serial.println(
            "PHASE 6 DISTRIBUTED SET: PASS"
        );
    } else {
        Serial.println();
        Serial.println(
            "PHASE 6 DISTRIBUTED SET: FAIL"
        );
    }

    Serial.println(
        "========================================"
    );
}

void start_client_request()
{
    if (
    LOCAL_NODE_ID != node::C ||
    request_started
) {
    return;
}

    /*
     * C must choose a key whose current primary is A.
     */
    if (
        !find_key_owned_by_node(
            node::A
        )
    ) {
        Serial.println(
            "NO KEY FOUND WITH PRIMARY A"
        );
        request_started = true;
        return;
    }

    static const std::uint8_t value[] = {
        'd', 'i', 's', 't', 'r', 'i', 'b', 'u', 't', 'e', 'd'
    };

    active_request =
        request_id_generator.next();

    if (!active_request.valid()) {
        Serial.println(
            "REQUEST ID GENERATION: FAIL"
        );

        request_started = true;
        return;
    }

    if (
        !build_set_request(
            active_request,
            test_key,
            test_key_length,
            value,
            sizeof(value),
            active_packet,
            sizeof(active_packet),
            active_packet_length
        )
    ) {
        Serial.println(
            "SET_REQ BUILD: FAIL"
        );

        request_started = true;
        return;
    }

    bool initial_send_ok = false;

    const request::Result submit_result =
        request_manager.submit(
            active_request,
            node::A,
            active_packet,
            active_packet_length,
            static_cast<std::uint64_t>(millis()),
            reliable_send,
            &sender_context,
            initial_send_ok
        );

    request_started = true;

    if (
        submit_result !=
        request::Result::OK
    ) {
        Serial.print(
            "REQUEST SUBMIT: FAIL ("
        );

        Serial.print(
            static_cast<unsigned>(
                submit_result
            )
        );

        Serial.println(")");

        return;
    }

    Serial.println();
    Serial.println(
        "CLIENT: SET_REQ SUBMITTED"
    );

    Serial.print(
        "REQUEST ID: "
    );

    Serial.println(
        active_request.request_id
    );

    Serial.print(
        "INITIAL SEND: "
    );

    Serial.println(
        initial_send_ok
            ? "OK"
            : "TRANSPORT FAILURE"
    );
}

void process_incoming()
{
    transport::ReceivedFrame frame{};

    while (
        espnow_transport.receive(
            frame
        )
    ) {
        if (
    LOCAL_NODE_ID == node::A
) {
    process_server_request(
        frame
    );
} else if (
    LOCAL_NODE_ID == node::C
) {
    process_requester_response(
        frame
    );
}
    }
}

bool initialize_system()
{
    Serial.println();
    Serial.println(
        "========================================"
    );
    Serial.println(
        "PHASE 6 - DISTRIBUTED SET INTEGRATION"
    );
    Serial.println(
        "========================================"
    );

    Serial.print(
        "LOCAL NODE: "
    );

    Serial.println(
        static_cast<unsigned>(
            LOCAL_NODE_ID
        )
    );

    /*
     * --------------------------------------------------
     * Persistent storage
     * --------------------------------------------------
     */
    Serial.println(
        "Initializing StorageManager..."
    );

    if (
        storage_manager.begin() !=
        storage::Result::OK
    ) {
        Serial.println(
            "StorageManager: FAIL"
        );
        return false;
    }

    storage::Hlc persisted_hlc = 0;

    if (
        storage_manager.max_persisted_hlc(
            persisted_hlc
        ) != storage::Result::OK
    ) {
        Serial.println(
            "Max persisted HLC: FAIL"
        );
        return false;
    }

    Serial.print(
        "Max persisted HLC: "
    );

    Serial.println(
        static_cast<unsigned long long>(
            persisted_hlc
        )
    );

    /*
     * --------------------------------------------------
     * HLC
     * --------------------------------------------------
     */
    if (
        !hlc_clock.begin(
            persisted_hlc
        )
    ) {
        Serial.println(
            "HLC begin: FAIL"
        );
        return false;
    }

    Serial.println(
        "HLC: READY"
    );

    /*
     * --------------------------------------------------
     * Membership
     *
     * For this integration test we explicitly seed the
     * A/C membership view. Heartbeat itself was already
     * validated in the previous phase.
     * --------------------------------------------------
     */
    if (
        !membership_manager.begin(
            static_cast<std::uint64_t>(millis())
        )
    ) {
        Serial.println(
            "Membership begin: FAIL"
        );
        return false;
    }

    membership_manager.heartbeat_received(
        TEST_PEER_NODE_ID,
        static_cast<std::uint64_t>(millis())
    );

    Serial.println(
        "Membership: LOCAL + PEER ALIVE"
    );

    /*
     * --------------------------------------------------
     * Placement
     * --------------------------------------------------
     */
    if (
        !placement_resolver.begin()
    ) {
        Serial.println(
            "Placement begin: FAIL"
        );
        return false;
    }

    Serial.println(
        "Placement: READY"
    );

    /*
     * --------------------------------------------------
     * KV engine
     * --------------------------------------------------
     */
    if (
        !kv_engine.begin()
    ) {
        Serial.println(
            "KV Engine begin: FAIL"
        );
        return false;
    }

    Serial.println(
        "KV Engine: READY"
    );

    /*
     * --------------------------------------------------
     * ESP-NOW transport
     * --------------------------------------------------
     */
    if (
        espnow_transport.begin(
            LOCAL_NODE_ID
        ) != transport::Result::OK
    ) {
        Serial.println(
            "ESP-NOW transport: FAIL"
        );
        return false;
    }

    Serial.println(
        "ESP-NOW transport: READY"
    );

    /*
     * --------------------------------------------------
     * Reliability / deduplication
     * --------------------------------------------------
     */
    if (
        request_manager.begin() !=
        request::Result::OK
    ) {
        Serial.println(
            "RequestManager: FAIL"
        );
        return false;
    }

    if (
        !dedup_manager.begin()
    ) {
        Serial.println(
            "DedupManager: FAIL"
        );
        return false;
    }

    if (
        !request_id_generator.begin(
            LOCAL_NODE_ID,
            static_cast<std::uint32_t>(
                esp_random()
            )
        )
    ) {
        Serial.println(
            "Request ID Generator: FAIL"
        );
        return false;
    }

    Serial.println(
        "Reliability + Dedup: READY"
    );

    Serial.println();
    Serial.println(
        "SYSTEM READY"
    );

    return true;
}

{
    Serial.begin(115200);

    delay(1000);

    system_ready =
        initialize_system();

    if (!system_ready) {
        Serial.println();
        Serial.println(
            "SYSTEM INITIALIZATION FAILED"
        );
        return;
    }

    client_start_time_ms =
        static_cast<std::uint64_t>(millis());
}

void loop()
{
    if (!system_ready) {
        delay(100);
        return;
    }

    const std::uint64_t now =
        static_cast<std::uint64_t>(millis());

    process_incoming();

    dedup_manager.expire(
        now
    );

    request_manager.tick(
        now
    );

    if (
        LOCAL_NODE_ID == node::C &&
        !request_started &&
        now >=
            client_start_time_ms +
            CLIENT_START_DELAY_MS
    ) {
        start_client_request();
    }

    if (
        LOCAL_NODE_ID == node::C &&
        request_finished
    ) {
        print_final_result();
    }

    delay(1);

bool is_valid_node_id(NodeId node_id) {
    return node_id >= NODE_ID_MIN &&
           node_id <= NODE_ID_MAX;
}

bool is_valid_destination(NodeId node_id) {
    return is_valid_node_id(node_id) ||
           node_id == NODE_ID_BROADCAST;
}

bool is_known_message_type(MessageType type) {
    const auto value = static_cast<std::uint8_t>(type);

    return value >= 0x01 &&
           value <= 0x0B;
}

bool has_valid_reserved_flags(std::uint8_t flags) {
    return (flags & FLAG_RESERVED_MASK) == 0;
}

Result serialize_set_request(
    const SetRequestPayload& payload,
    std::uint8_t* out,
    std::size_t out_capacity,
    std::size_t& bytes_written
) {
    bytes_written = 0;

    if (out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload.key.len > KEY_MAX ||
        payload.value.len > VALUE_MAX) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.key.len > 0 && payload.key.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload.value.len > 0 && payload.value.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    const std::size_t required =
        1 + payload.key.len +
        1 + payload.value.len;

    if (required > MAX_PAYLOAD_SIZE) {
        return Result::INVALID_PAYLOAD;
    }

    if (out_capacity < required) {
        return Result::BUFFER_TOO_SMALL;
    }

    out[0] = payload.key.len;

    if (payload.key.len > 0) {
        std::memcpy(
            out + 1,
            payload.key.data,
            payload.key.len
        );
    }

    const std::size_t value_len_offset =
        1 + payload.key.len;

    out[value_len_offset] = payload.value.len;

    if (payload.value.len > 0) {
        std::memcpy(
            out + value_len_offset + 1,
            payload.value.data,
            payload.value.len
        );
    }

    bytes_written = required;

    return Result::OK;
}

Result deserialize_set_request(
    const std::uint8_t* data,
    std::size_t payload_len,
    SetRequestPayload& payload
) {
    payload.key = {};
    payload.value = {};

    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload_len < 2 ||
        payload_len > MAX_PAYLOAD_SIZE) {
        return Result::INVALID_PAYLOAD;
    }

    const std::uint8_t key_len = data[0];

    if (key_len > KEY_MAX) {
        return Result::INVALID_PAYLOAD;
    }

    const std::size_t value_len_offset =
        1 + key_len;

    if (value_len_offset >= payload_len) {
        return Result::INVALID_PAYLOAD;
    }

    const std::uint8_t value_len =
        data[value_len_offset];

    if (value_len > VALUE_MAX) {
        return Result::INVALID_PAYLOAD;
    }

    const std::size_t expected_len =
        1 + key_len +
        1 + value_len;

    if (expected_len != payload_len) {
        return Result::INVALID_PAYLOAD;
    }

    payload.key = {
        data + 1,
        key_len
    };

    payload.value = {
        data + value_len_offset + 1,
        value_len
    };

    return Result::OK;
}








namespace {

class PayloadWriter {
public:
    PayloadWriter(
        std::uint8_t* data,
        std::size_t capacity
    )
        : data_(data), capacity_(capacity), position_(0) {}

    bool write_u8(std::uint8_t value) {
        if (position_ >= capacity_) {
            return false;
        }

        data_[position_++] = value;
        return true;
    }

    bool write_bytes(
        const std::uint8_t* data,
        std::size_t len
    ) {
        if (len > capacity_ - position_) {
            return false;
        }

        if (len > 0) {
            std::memcpy(
                data_ + position_,
                data,
                len
            );
        }

        position_ += len;
        return true;
    }

    bool write_byte_view(ByteView view) {
        if (!write_u8(view.len)) {
            return false;
        }

        return write_bytes(view.data, view.len);
    }

    std::size_t size() const {
        return position_;
    }

private:
    std::uint8_t* data_;
    std::size_t capacity_;
    std::size_t position_;
};

class PayloadReader {
public:
    PayloadReader(
        const std::uint8_t* data,
        std::size_t length
    )
        : data_(data), length_(length), position_(0) {}

    bool read_u8(std::uint8_t& value) {
        if (position_ >= length_) {
            return false;
        }

        value = data_[position_++];
        return true;
    }

    bool read_byte_view(ByteView& view) {
        std::uint8_t len = 0;

        if (!read_u8(len)) {
            return false;
        }

        if (len > length_ - position_) {
            return false;
        }

        view = {
            data_ + position_,
            len
        };

        position_ += len;

        return true;
    }

    bool finished() const {
        return position_ == length_;
    }




private:
    const std::uint8_t* data_;
    std::size_t length_;
    std::size_t position_;
};

Result serialize_replication_request(
    const ReplicationRequestPayload& payload,
    std::uint8_t* out,
    std::size_t out_capacity,
    std::size_t& bytes_written
) {
    bytes_written = 0;

    if (out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload.key.len > KEY_MAX ||
        payload.value.len > VALUE_MAX) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.key.len > 0 &&
        payload.key.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload.value.len > 0 &&
        payload.value.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    const std::size_t required =
        static_cast<std::size_t>(1) + payload.key.len +
        static_cast<std::size_t>(1) + payload.value.len;

    if (required > MAX_PAYLOAD_SIZE) {
        return Result::INVALID_PAYLOAD;
    }

    if (out_capacity < required) {
        return Result::BUFFER_TOO_SMALL;
    }

    PayloadWriter writer(out, out_capacity);

    if (!writer.write_byte_view(payload.key) ||
        !writer.write_byte_view(payload.value)) {
        return Result::BUFFER_TOO_SMALL;
    }

    bytes_written = writer.size();

    return Result::OK;
}

Result deserialize_replication_request(
    const std::uint8_t* data,
    std::size_t payload_len,
    ReplicationRequestPayload& payload
) {
    payload = {};

    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload_len > MAX_PAYLOAD_SIZE) {
        return Result::INVALID_PAYLOAD;
    }

    PayloadReader reader(data, payload_len);

    if (!reader.read_byte_view(payload.key) ||
        !reader.read_byte_view(payload.value)) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.key.len > KEY_MAX ||
        payload.value.len > VALUE_MAX ||
        !reader.finished()) {
        return Result::INVALID_PAYLOAD;
    }

    return Result::OK;
}

    const SetAckPayload& payload,
    std::uint8_t* out,
    std::size_t out_capacity,
    std::size_t& bytes_written

    bytes_written = 0;

    if (out == nullptr) {
        return Result::INVALID_ARGUMENT;

    if (payload.key.len > KEY_MAX) {

    }

    if (payload.key.len > 0 &&

Result deserialize_set_ack(
    const std::uint8_t* data,
    std::size_t payload_len,
    SetAckPayload& payload
) {
    payload = {};

    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    PayloadReader reader(data, payload_len);

    std::uint8_t status = 0;

    if (!reader.read_u8(status) ||
        !reader.read_byte_view(payload.key) ||
        payload.key.len > KEY_MAX ||
        !reader.finished()) {
        return Result::INVALID_PAYLOAD;
    }

    payload.status =
        static_cast<StatusCode>(status);

    return Result::OK;
}
Result serialize_get_response(

        return Result::BUFFER_TOO_SMALL;

    PayloadWriter writer(out, out_capacity);

    if (!writer.write_u8(
            static_cast<std::uint8_t>(payload.status)) ||
        !writer.write_byte_view(payload.key) ||
        !writer.write_byte_view(payload.value)) {
        return Result::BUFFER_TOO_SMALL;
    }

    bytes_written = writer.size();
    return Result::OK;

Result deserialize_get_response(
    const std::uint8_t* data,
    std::size_t payload_len,
    GetResponsePayload& payload
) {
    payload = {};

    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    PayloadReader reader(data, payload_len);

    std::uint8_t status = 0;

    if (!reader.read_u8(status) ||
        !reader.read_byte_view(payload.key) ||
        !reader.read_byte_view(payload.value)) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.key.len > KEY_MAX ||
        payload.value.len > VALUE_MAX ||
        !reader.finished()) {
        return Result::INVALID_PAYLOAD;
    }

    payload.status =
        static_cast<StatusCode>(status);

    if (payload.status != StatusCode::OK &&
        payload.value.len != 0) {
        return Result::INVALID_PAYLOAD;
    }

    return Result::OK;
}

namespace {

void write_u16_be(std::uint8_t* out, std::uint16_t value) {
    out[0] = static_cast<std::uint8_t>((value >> 8) & 0xFF);
    out[1] = static_cast<std::uint8_t>(value & 0xFF);
}

void write_u32_be(std::uint8_t* out, std::uint32_t value) {
    out[0] = static_cast<std::uint8_t>((value >> 24) & 0xFF);
    out[1] = static_cast<std::uint8_t>((value >> 16) & 0xFF);
    out[2] = static_cast<std::uint8_t>((value >> 8) & 0xFF);
    out[3] = static_cast<std::uint8_t>(value & 0xFF);
}

void write_u64_be(std::uint8_t* out, std::uint64_t value) {
    for (int i = 0; i < 8; ++i) {
        out[i] = static_cast<std::uint8_t>(
            (value >> (56 - 8 * i)) & 0xFF
        );
    }
}

std::uint16_t read_u16_be(const std::uint8_t* data) {
    return static_cast<std::uint16_t>(
        (static_cast<std::uint16_t>(data[0]) << 8) |
        static_cast<std::uint16_t>(data[1])
    );
}

std::uint32_t read_u32_be(const std::uint8_t* data) {
    return
        (static_cast<std::uint32_t>(data[0]) << 24) |
        (static_cast<std::uint32_t>(data[1]) << 16) |
        (static_cast<std::uint32_t>(data[2]) << 8) |
        static_cast<std::uint32_t>(data[3]);
}

std::uint64_t read_u64_be(const std::uint8_t* data) {
    std::uint64_t value = 0;

    for (int i = 0; i < 8; ++i) {
        value |= static_cast<std::uint64_t>(data[i])
                 << (56 - 8 * i);
    }

    return value;
}

Result serialize_version_table(
    const VersionTablePayload& payload,
    std::uint8_t* out,
    std::size_t out_capacity,
    std::size_t& bytes_written
) {
    bytes_written = 0;

    if (out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload.entry_count > VERSION_TABLE_MAX_ENTRIES) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.entry_count > 0 && payload.entries == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload.chunk_count == 0 ||
        payload.chunk_index >= payload.chunk_count) {
        return Result::INVALID_PAYLOAD;
    }

    std::size_t required = 9;

    for (std::size_t i = 0; i < payload.entry_count; ++i) {
        const VersionTableEntry& entry = payload.entries[i];

        if (entry.key.len > KEY_MAX) {
            return Result::INVALID_PAYLOAD;
        }

        if (entry.key.len > 0 &&
            entry.key.data == nullptr) {
            return Result::INVALID_ARGUMENT;
        }

        if (!is_valid_node_id(entry.version_origin_id)) {
            return Result::INVALID_PAYLOAD;
        }

        required +=
            static_cast<std::size_t>(11) +
            entry.key.len;
    }

    if (required > MAX_PAYLOAD_SIZE) {
        return Result::INVALID_PAYLOAD;
    }

    if (out_capacity < required) {
        return Result::BUFFER_TOO_SMALL;
    }

    write_u32_be(out, payload.sync_id);
    write_u16_be(out + 4, payload.chunk_index);
    write_u16_be(out + 6, payload.chunk_count);

    out[8] = payload.entry_count;

    std::size_t offset = 9;

    for (std::size_t i = 0; i < payload.entry_count; ++i) {
        const VersionTableEntry& entry = payload.entries[i];

        out[offset++] = entry.entry_flags;
        out[offset++] = entry.key.len;

        if (entry.key.len > 0) {
            std::memcpy(
                out + offset,
                entry.key.data,
                entry.key.len
            );

            offset += entry.key.len;
        }

        write_u64_be(out + offset, entry.version_hlc);
        offset += 8;

        out[offset++] = entry.version_origin_id;
    }

    bytes_written = offset;

    return Result::OK;
}


Result deserialize_version_table(
    const std::uint8_t* data,
    std::size_t payload_len,
    VersionTablePayload& payload,
    VersionTableEntry* entries_out,
    std::size_t entries_capacity
) {
    payload = {};

    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload_len < 9 ||
        payload_len > MAX_PAYLOAD_SIZE) {
        return Result::INVALID_PAYLOAD;
    }

    payload.sync_id = read_u32_be(data);
    payload.chunk_index = read_u16_be(data + 4);
    payload.chunk_count = read_u16_be(data + 6);
    payload.entry_count = data[8];

    if (payload.chunk_count == 0 ||
        payload.chunk_index >= payload.chunk_count) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.entry_count > VERSION_TABLE_MAX_ENTRIES) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.entry_count > entries_capacity) {
        return Result::BUFFER_TOO_SMALL;
    }

    if (payload.entry_count > 0 &&
        entries_out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    std::size_t offset = 9;

    for (std::size_t i = 0;
         i < payload.entry_count;
         ++i) {

        if (payload_len - offset < 11) {
            return Result::INVALID_PAYLOAD;
        }

        VersionTableEntry& entry = entries_out[i];

        entry.entry_flags = data[offset++];

        const std::uint8_t key_len = data[offset++];

        if (key_len > KEY_MAX) {
            return Result::INVALID_PAYLOAD;
        }

        if (payload_len - offset <
            static_cast<std::size_t>(key_len) + 9) {
            return Result::INVALID_PAYLOAD;
        }

        entry.key = {
            data + offset,
            key_len
        };

        offset += key_len;

        entry.version_hlc =
            read_u64_be(data + offset);

        offset += 8;

        entry.version_origin_id = data[offset++];

        if (!is_valid_node_id(entry.version_origin_id)) {
            return Result::INVALID_PAYLOAD;
        }
    }

    if (offset != payload_len) {
        return Result::INVALID_PAYLOAD;
    }

    payload.entries = entries_out;

    return Result::OK;
}

Result serialize_header(
    const PacketHeader& header,
    std::uint8_t* out,
    std::size_t out_capacity
) {
    if (out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (out_capacity < HEADER_SIZE) {
        return Result::BUFFER_TOO_SMALL;
    }

    if (header.protocol_version != PROTOCOL_VERSION ||
        header.header_len != HEADER_SIZE ||
        header.payload_len > MAX_PAYLOAD_SIZE ||
        header.ttl != 0 ||
        !is_valid_node_id(header.origin_node_id) ||
        !is_valid_node_id(header.sender_node_id) ||
        !is_valid_destination(header.destination_node_id) ||
        !is_known_message_type(header.msg_type) ||
        !has_valid_reserved_flags(header.flags)) {
        return Result::INVALID_PACKET;
    }

    out[0] = header.protocol_version;
    out[1] = static_cast<std::uint8_t>(header.msg_type);
    out[2] = header.flags;
    out[3] = header.header_len;

    write_u32_be(out + 4, header.request_id);

    out[8] = header.origin_node_id;
    out[9] = header.sender_node_id;
    out[10] = header.destination_node_id;
    out[11] = header.ttl;

    write_u64_be(out + 12, header.hlc);
    write_u16_be(out + 20, header.payload_len);

    return Result::OK;
}
Result deserialize_header(
    const std::uint8_t* data,
    std::size_t packet_len,
    PacketHeader& header
) {
    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (packet_len < HEADER_SIZE) {
        return Result::INVALID_PACKET;
    }

    header.protocol_version = data[0];
    header.msg_type =
        static_cast<MessageType>(data[1]);
    header.flags = data[2];
    header.header_len = data[3];

    header.request_id = read_u32_be(data + 4);

    header.origin_node_id = data[8];
    header.sender_node_id = data[9];
    header.destination_node_id = data[10];
    header.ttl = data[11];

    header.hlc = read_u64_be(data + 12);
    header.payload_len = read_u16_be(data + 20);

    if (header.protocol_version != PROTOCOL_VERSION ||
        header.header_len != HEADER_SIZE ||
        header.payload_len > MAX_PAYLOAD_SIZE ||
        packet_len != HEADER_SIZE + header.payload_len ||
        header.ttl != 0 ||
        !is_valid_node_id(header.origin_node_id) ||
        !is_valid_node_id(header.sender_node_id) ||
        !is_valid_destination(header.destination_node_id) ||
        !is_known_message_type(header.msg_type) ||
        !has_valid_reserved_flags(header.flags)) {
        return Result::INVALID_PACKET;
    }

    return Result::OK;
}

Result validate_message_header(
    const PacketHeader& header
) {
    switch (header.msg_type) {
        case MessageType::SET_REQ:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::SET_ACK:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::GET_REQ:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::GET_RESP:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::DEL_REQ:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::DEL_ACK:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::REPL_REQ:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::REPL_ACK:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::HEARTBEAT:
            if (header.destination_node_id != NODE_ID_BROADCAST) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::VERSION_TABLE:
            if (header.destination_node_id != NODE_ID_BROADCAST) {
                return Result::INVALID_PACKET;
            }
            break;

        case MessageType::RESYNC_DATA:
            if (!is_valid_node_id(header.destination_node_id)) {
                return Result::INVALID_PACKET;
            }
            break;

        default:
            return Result::INVALID_MESSAGE;
    }

    return Result::OK;
}
Result decode_message(
    const PacketView& packet,
    DecodedMessage& message
) {
    message = {};

    Result result = validate_message_header(packet.header);

    if (result != Result::OK) {
        return result;
    }

    message.header = packet.header;

    switch (packet.header.msg_type) {
        case MessageType::SET_REQ:
            return deserialize_set_request(
                packet.payload.data,
                packet.payload.len,
                message.payload.set_request
            );

        case MessageType::SET_ACK:
            return deserialize_set_ack(
                packet.payload.data,
                packet.payload.len,
                message.payload.set_ack
            );

        case MessageType::GET_REQ:
            return deserialize_get_request(
                packet.payload.data,
                packet.payload.len,
                message.payload.get_request
            );

        case MessageType::GET_RESP:
            return deserialize_get_response(
                packet.payload.data,
                packet.payload.len,
                message.payload.get_response
            );

        case MessageType::DEL_REQ:
            return deserialize_delete_request(
                packet.payload.data,
                packet.payload.len,
                message.payload.delete_request
            );

        case MessageType::DEL_ACK:
            return deserialize_delete_ack(
                packet.payload.data,
                packet.payload.len,
                message.payload.delete_ack
            );

        case MessageType::REPL_REQ:
            return deserialize_replication_request(
                packet.payload.data,
                packet.payload.len,
                message.payload.replication_request
            );

        case MessageType::REPL_ACK:
            return deserialize_replication_ack(
                packet.payload.data,
                packet.payload.len,
                message.payload.replication_ack
            );

        case MessageType::HEARTBEAT:
            if (packet.payload.len != 0) {
                return Result::INVALID_PAYLOAD;
            }

            return Result::OK;

        case MessageType::VERSION_TABLE:
            return deserialize_version_table(
                packet.payload.data,
                packet.payload.len,
                message.payload.version_table,
                message.version_table_entries,
                VERSION_TABLE_MAX_ENTRIES
            );

        case MessageType::RESYNC_DATA:
            return deserialize_resync_data(
                packet.payload.data,
                packet.payload.len,
                message.payload.resync_data
            );

        default:
            return Result::INVALID_MESSAGE;
    }
}



Result serialize_packet(
    const PacketHeader& header,
    ByteView payload,
    std::uint8_t* out,
    std::size_t out_capacity,
    std::size_t& bytes_written
) {
    bytes_written = 0;

    if (out == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (payload.len > MAX_PAYLOAD_SIZE) {
        return Result::INVALID_PAYLOAD;
    }

    if (payload.len > 0 && payload.data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    if (header.payload_len != payload.len) {
        return Result::INVALID_PACKET;
    }

    const std::size_t required =
        HEADER_SIZE + payload.len;

    if (out_capacity < required) {
        return Result::BUFFER_TOO_SMALL;
    }

    Result result = serialize_header(
        header,
        out,
        out_capacity
    );

    if (result != Result::OK) {
        return result;
    }

    if (payload.len > 0) {
        std::memcpy(
            out + HEADER_SIZE,
            payload.data,
            payload.len
        );
    }

    bytes_written = required;

    return Result::OK;
}

Result deserialize_packet(
    const std::uint8_t* data,
    std::size_t packet_len,
    PacketView& packet
) {
    packet = {};

    if (data == nullptr) {
        return Result::INVALID_ARGUMENT;
    }

    Result result = deserialize_header(
        data,
        packet_len,
        packet.header
    );

    if (result != Result::OK) {
        return result;
    }

    packet.payload = {
        data + HEADER_SIZE,
        static_cast<std::uint8_t>(packet.header.payload_len)
    };

    return Result::OK;
}

protocol::MessageType ResponseBuilder::response_type_for(
    protocol::MessageType request_type
)
{
    switch (request_type) {
        case protocol::MessageType::SET_REQ:
            return protocol::MessageType::SET_ACK;

        case protocol::MessageType::GET_REQ:
            return protocol::MessageType::GET_RESP;

        case protocol::MessageType::DEL_REQ:
            return protocol::MessageType::DEL_ACK;

        default:
            return request_type;
    }
}

protocol::StatusCode ResponseBuilder::to_protocol_status(
    Result result
)
{
    switch (result) {
        case Result::OK:
            return protocol::StatusCode::OK;

        case Result::NOT_FOUND:
            return protocol::StatusCode::NOT_FOUND;

        case Result::INVALID_REQUEST:
            return protocol::StatusCode::INVALID_REQUEST;

        case Result::KEY_TOO_LARGE:
            return protocol::StatusCode::KEY_TOO_LARGE;

        case Result::VALUE_TOO_LARGE:
            return protocol::StatusCode::VALUE_TOO_LARGE;

        case Result::STORAGE_ERROR:
            return protocol::StatusCode::STORAGE_ERROR;

        case Result::NOT_RESPONSIBLE:
            return protocol::StatusCode::NOT_RESPONSIBLE;

        case Result::STALE_VERSION:
            return protocol::StatusCode::STALE_VERSION;

        case Result::NOT_READY:
            return protocol::StatusCode::STORAGE_ERROR;

        default:
            return protocol::StatusCode::INVALID_REQUEST;
    }
}

bool ResponseBuilder::build_payload(
    protocol::MessageType response_type,
    const OperationResponse& response,
    std::uint8_t* output,
    std::size_t output_capacity,
    std::size_t& payload_length
)
{
    payload_length = 0;

    const protocol::StatusCode status =
        to_protocol_status(response.result);

    const protocol::KeyView key{
        response.key,
        response.key_length
    };

    switch (response_type) {
        case protocol::MessageType::SET_ACK: {
            const protocol::SetAckPayload payload{
                status,
                key
            };

            return protocol::serialize_set_ack(
                payload,
                output,
                output_capacity,
                payload_length
            ) == protocol::Result::OK;
        }

        case protocol::MessageType::GET_RESP: {
            const protocol::ValueView value{
                response.value,
                static_cast<std::uint8_t>(
                    response.value_length
                )
            };

            const protocol::GetResponsePayload payload{
                status,
                key,
                value
            };

            return protocol::serialize_get_response(
                payload,
                output,
                output_capacity,
                payload_length
            ) == protocol::Result::OK;
        }

        case protocol::MessageType::DEL_ACK: {
            const protocol::DeleteAckPayload payload{
                status,
                key
            };

            return protocol::serialize_delete_ack(
                payload,
                output,
                output_capacity,
                payload_length
            ) == protocol::Result::OK;
        }

        default:
            return false;
    }
}

bool ResponseBuilder::build(
    const protocol::PacketHeader& request_header,
    node::Id local_node_id,
    protocol::Hlc response_hlc,
    const OperationResponse& response,
    std::uint8_t* output,
    std::size_t output_capacity,
    std::size_t& output_length
)
{
    output_length = 0;

    if (
        output == nullptr ||
        output_capacity < protocol::HEADER_SIZE
    ) {
        return false;
    }

    if (
        local_node_id < node::MIN ||
        local_node_id > node::MAX
    ) {
        return false;
    }

    const protocol::MessageType response_type =
        response_type_for(
            request_header.msg_type
        );

    if (
        response_type != protocol::MessageType::SET_ACK &&
        response_type != protocol::MessageType::GET_RESP &&
        response_type != protocol::MessageType::DEL_ACK
    ) {
        return false;
    }

    std::uint8_t payload[
        protocol::MAX_PAYLOAD_SIZE
    ]{};

    std::size_t payload_length = 0;

    if (
        !build_payload(
            response_type,
            response,
            payload,
            sizeof(payload),
            payload_length
        )
    ) {
        return false;
    }

    if (
        payload_length >
        protocol::MAX_PAYLOAD_SIZE
    ) {
        return false;
    }

    const protocol::PacketHeader response_header{
        protocol::PROTOCOL_VERSION,
        response_type,
        0,
        protocol::HEADER_SIZE,

        request_header.request_id,

        request_header.origin_node_id,
        local_node_id,
        request_header.origin_node_id,

        0,

        response_hlc,

        static_cast<std::uint16_t>(
            payload_length
        )
    };

    return protocol::serialize_packet(
        response_header,
        protocol::ByteView{
            payload,
            static_cast<std::uint8_t>(
                payload_length
            )
        },
        output,
        output_capacity,
        output_length
    ) == protocol::Result::OK;
}

constexpr std::uint64_t DEFAULT_OVERALL_DEADLINE_MS = 5000;

struct TimingConfig {
    std::uint64_t initial_timeout_ms =
        DEFAULT_INITIAL_TIMEOUT_MS;

    std::uint64_t max_retry_timeout_ms =
        DEFAULT_MAX_RETRY_TIMEOUT_MS;

    std::uint8_t max_retransmissions =
        DEFAULT_MAX_RETRANSMISSIONS;

    std::uint64_t overall_deadline_ms =
        DEFAULT_OVERALL_DEADLINE_MS;
};

enum class State : std::uint8_t {
    FREE = 0,
    PENDING,
    COMPLETED,
    FAILED
};

enum class Result : std::uint8_t {
    OK = 0,
    INVALID_ARGUMENT,
    NOT_READY,
    TABLE_FULL,
    DUPLICATE,
    UNKNOWN_REQUEST,
    ALREADY_TERMINAL,
    BUFFER_TOO_SMALL
};

enum class SendFailurePolicy : std::uint8_t {
    RETRY = 0
};

using SendFn = bool (*)(
    void* context,
    node::Id destination,
    const std::uint8_t* packet,
    std::size_t packet_length,
    bool retry
);

struct CompletionView {
    const std::uint8_t* data;
    std::size_t length;
};

struct RequestInfo {
    Identity identity;
    node::Id destination;
    State state;
    std::uint8_t retry_count;
};

class Manager {
public:
    explicit Manager(
        TimingConfig config = TimingConfig{}
    );

    Result begin();

    Result submit(
        Identity identity,
        node::Id destination,
        const std::uint8_t* packet,
        std::size_t packet_length,
        std::uint64_t now_ms,
        SendFn send_fn,
        void* send_context,
        bool& initial_send_ok
    );

    void tick(
        std::uint64_t now_ms
    );

    Result complete(
        Identity identity,
        const std::uint8_t* response,
        std::size_t response_length
    );

    Result get_state(
        Identity identity,
        State& state
    ) const;

    Result get_info(
        Identity identity,
        RequestInfo& info
    ) const;

    Result get_completion(
        Identity identity,
        std::uint8_t* output,
        std::size_t output_capacity,
        std::size_t& output_length
    ) const;

    Result release(
        Identity identity
    );

    std::size_t pending_count() const;

    bool ready() const;

private:
    struct Entry {
        Identity identity{};
        node::Id destination = node::INVALID;

        std::uint8_t packet[MAX_PACKET_SIZE]{};
        std::uint8_t packet_length = 0;

        std::uint8_t response[MAX_PACKET_SIZE]{};
        std::uint8_t response_length = 0;

        std::uint64_t created_at_ms = 0;
        std::uint64_t deadline_ms = 0;
        std::uint64_t next_retry_ms = 0;
        std::uint64_t current_retry_timeout_ms = 0;

        std::uint8_t retry_count = 0;

        State state = State::FREE;

        SendFn send_fn = nullptr;
        void* send_context = nullptr;
    };

    int find(
        Identity identity
    ) const;

    int find_free() const;

    std::uint64_t retry_delay(
        std::uint64_t current_timeout
    ) const;

    void fail_entry(
        Entry& entry
    );

    TimingConfig config_;
    Entry entries_[MAX_PENDING_REQUESTS]{};
    bool initialized_ = false;
};
